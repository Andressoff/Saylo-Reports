function showScreen(screenId) {
    if (isRecording) {
        stopRecording()
    }

    Object.values({
        homeScreen: elements.homeScreen,
        lessonScreen: elements.lessonScreen,
        finishScreen: elements.finishScreen,
        parentsScreen: elements.parentsScreen
    }).forEach(screen => {
        screen.classList.remove('active')
    })

    const target = document.getElementById(screenId)

    if (!target) {
        console.error('Экран не найден:', screenId)
        return
    }

    target.classList.add('active')
    currentScreen = screenId

    const isParentsMode = screenId === 'parentsScreen'

    document.body.classList.toggle('parents-mode', isParentsMode)

    if (gameEngine?.initialized) {
        if (isParentsMode) {
            gameEngine.pause()
        } else {
            gameEngine.resume()
        }
    }
}

function stopSpeech() {
    if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel()
    }
}

function speak(text) {
    if (!text || !('speechSynthesis' in window)) {
        return
    }

    stopSpeech()

    const utterance = new SpeechSynthesisUtterance(text)
    utterance.lang = 'ru-RU'
    utterance.rate = 0.85
    utterance.pitch = 1.05
    utterance.volume = 1

    window.speechSynthesis.speak(utterance)
}

function flyBeeToLesson() {

    const bee = gameEngine?.scene?.beeCharacter
    const camera = gameEngine?.camera
    const canvas = gameEngine?.app?.canvas

    const card = document.querySelector(
        '#lessonScreen .task-card'
    )

    if (!bee || !camera || !canvas || !card) {
        console.warn('[Saylo Bee] Не удалось определить место для полёта.')
        return false
    }

    const canvasRect = canvas.getBoundingClientRect()
    const cardRect = card.getBoundingClientRect()

    if (
        canvasRect.width <= 0 ||
        canvasRect.height <= 0 ||
        cardRect.width <= 0 ||
        cardRect.height <= 0
    ) {
        return false
    }

    const cssPerWorldX =
        canvasRect.width / camera.viewportWidth

    const cssPerWorldY =
        canvasRect.height / camera.viewportHeight

    const baseWidth = bee.baseWidth * camera.scale * cssPerWorldX
    const baseHeight = bee.baseHeight * camera.scale * cssPerWorldY

    const leftSpace = Math.max(
        0,
        cardRect.left - canvasRect.left
    )

    const rightSpace = Math.max(
        0,
        canvasRect.right - cardRect.right
    )

    const topSpace = Math.max(
        0,
        cardRect.top - canvasRect.top
    )

    const bottomSpace = Math.max(
        0,
        canvasRect.bottom - cardRect.bottom
    )

    let screenX
    let screenY

    const horizontalSpace = Math.max(leftSpace, rightSpace)

    if (horizontalSpace >= 135) {

        // Выбираем наиболее свободную сторону карточки.

        const useLeft = leftSpace >= rightSpace
        const available = useLeft ? leftSpace : rightSpace

        const scale = Math.max(
            0.38,
            Math.min(0.9, (available - 24) / baseWidth)
        )

        bee.setVisualScale(scale)

        const width = baseWidth * scale
        const height = baseHeight * scale

        screenX = useLeft
            ? cardRect.left - width / 2 - 12
            : cardRect.right + width / 2 + 12

        const minY = canvasRect.top + height / 2 + 8
        const maxY = canvasRect.bottom - height / 2 - 8

        screenY = minY <= maxY
            ? Math.max(minY, Math.min(maxY, cardRect.top + cardRect.height / 2))
            : canvasRect.top + canvasRect.height / 2

    } else {

        // На узком экране размещаем Шмелёнка над карточкой или под ней.

        const useTop = topSpace >= bottomSpace
        const available = useTop ? topSpace : bottomSpace

        const scale = Math.max(
            0.38,
            Math.min(0.8, (available - 20) / baseHeight)
        )

        bee.setVisualScale(scale)

        const width = baseWidth * scale
        const height = baseHeight * scale

        screenX = canvasRect.left + canvasRect.width / 2

        screenY = useTop
            ? cardRect.top - height / 2 - 10
            : cardRect.bottom + height / 2 + 10

        const minX = canvasRect.left + width / 2 + 8
        const maxX = canvasRect.right - width / 2 - 8

        if (minX <= maxX) {
            screenX = Math.max(minX, Math.min(maxX, screenX))
        }

        const minY = canvasRect.top + height / 2 + 8
        const maxY = canvasRect.bottom - height / 2 - 8

        if (minY <= maxY) {
            screenY = Math.max(minY, Math.min(maxY, screenY))
        }
    }

    // Перевод CSS-координат в координаты PixiJS.

    const logicalX =
        (screenX - canvasRect.left) /
        cssPerWorldX

    const logicalY =
        (screenY - canvasRect.top) /
        cssPerWorldY

    const worldPosition = camera.screenToWorld(
        logicalX,
        logicalY
    )

    bee.flyTo(
        worldPosition.x,
        worldPosition.y,
        {
            duration: 1.4,
            lift: Math.max(65, camera.viewportHeight * 0.10)
        }
    )

    return true
}

function renderTask() {
    const task = tasks[currentTaskIndex]

    if (!task) {
        showScreen('finishScreen')
        speak('Занятие завершено. Ты молодец!')
        return
    }

    const progress = ((currentTaskIndex + 1) / tasks.length) * 100

    elements.stepCounter.textContent =
        `${currentTaskIndex + 1} / ${tasks.length}`

    elements.lessonProgress.style.width = `${progress}%`

    elements.taskLabel.textContent = task.label
    elements.taskTitle.textContent = task.title
    elements.taskDescription.textContent = task.description
    elements.soundSymbol.textContent = task.symbol
    elements.taskHint.textContent = task.hint

    elements.taskFeedback.textContent = ''

    elements.choiceArea.classList.add('hidden')
    elements.recordArea.classList.add('hidden')
    elements.nextTaskButton.classList.add('hidden')

    elements.listenButton.classList.remove('hidden')

    elements.choiceButtons.replaceChildren()

    selectedAnswer = null

    if (task.type === 'choice') {
        elements.choiceArea.classList.remove('hidden')

        task.choices.forEach(choice => {
            const button = document.createElement('button')

            button.type = 'button'
            button.className = 'choice-button'
            button.textContent = choice

            button.addEventListener('click', () => {
                selectAnswer(choice, task.correct)
            })

            elements.choiceButtons.appendChild(button)
        })
    }

    if (task.type === 'record') {
        elements.recordArea.classList.remove('hidden')
        elements.listenButton.classList.remove('hidden')
        resetChildRecordButton()
    }

    if (task.type === 'finish') {
        elements.listenButton.classList.add('hidden')
        elements.nextTaskButton.classList.remove('hidden')
        elements.nextTaskButton.textContent = 'Завершить занятие'
    }

    if (task.type !== 'finish') {
        elements.nextTaskButton.textContent = 'Продолжить →'
    }

    speak(task.speech)
}

function selectAnswer(choice, correct) {
    if (selectedAnswer !== null) {
        return
    }

    selectedAnswer = choice

    const buttons = elements.choiceButtons.querySelectorAll('button')

    buttons.forEach(button => {
        button.disabled = true

        if (button.textContent === choice) {
            button.classList.add('selected')
        }
    })

    if (choice === correct) {
        elements.taskFeedback.textContent =
            'Отлично! Ты правильно выбрал ответ.'

        speak('Правильно! Молодец!')
    } else {
        elements.taskFeedback.textContent =
            'Хорошая попытка! Давай попробуем ещё раз в следующий раз.'

        speak('Хорошая попытка! Продолжаем.')
    }

    elements.nextTaskButton.classList.remove('hidden')
}

function nextTask() {
    const task = tasks[currentTaskIndex]

    if (task && task.type === 'finish') {
        showScreen('finishScreen')
        speak('Ты молодец! До новых встреч!')
        return
    }

    currentTaskIndex++

    if (currentTaskIndex >= tasks.length) {
        showScreen('finishScreen')
        speak('Занятие завершено. Спасибо!')
        return
    }

    renderTask()
}

async function startRecording(source) {
    if (isRecording) {
        return
    }

    try {
        stopSpeech()

        await recorder.start()

        isRecording = true
        currentRecordingSource = source

        if (source === 'child') {
            elements.recordButton.classList.add('recording')
            elements.recordButtonIcon.textContent = '⏹'
            elements.recordButtonText.textContent = 'Остановить запись'
            elements.recordStatus.textContent = 'Я слушаю тебя...'
        } else {
            elements.audioStatus.textContent = 'Идёт запись...'
            elements.parentRecordButton.disabled = true
            elements.parentStopButton.disabled = false
        }

    } catch (error) {
        console.error('Ошибка записи:', error)

        isRecording = false

        const message =
            'Не удалось включить микрофон. Проверь разрешение на его использование.'

        if (source === 'child') {
            elements.recordStatus.textContent = message
            resetChildRecordButton()
        } else {
            elements.audioStatus.textContent = message
            elements.parentRecordButton.disabled = false
            elements.parentStopButton.disabled = true
        }
    }
}

async function stopRecording() {
    if (!isRecording) {
        return
    }

    try {
        const blob = await recorder.stop()

        isRecording = false

        if (blob && blob.size > 0) {
            currentRecording = blob

            if (currentAudioUrl) {
                URL.revokeObjectURL(currentAudioUrl)
            }

            currentAudioUrl = URL.createObjectURL(blob)

            elements.audioPlayer.src = currentAudioUrl
            elements.audioPlayer.classList.remove('hidden')

            elements.transcribeButton.disabled = false
            elements.saveAudioButton.disabled = false
        }

    } catch (error) {
        console.error('Ошибка остановки записи:', error)
    } finally {
        isRecording = false

        if (currentRecordingSource === 'parent') {
            elements.audioStatus.textContent =
                currentRecording
                    ? 'Запись завершена. Её можно прослушать.'
                    : 'Запись остановлена.'

            elements.parentRecordButton.disabled = false
            elements.parentStopButton.disabled = true
        }

        currentRecordingSource = null
    }
}

async function transcribeAudio() {
    if (!currentRecording) {
        elements.transcriptText.textContent = 'Сначала сделай запись.'
        return
    }

    elements.transcribeButton.disabled = true
    elements.transcriptText.textContent = 'Распознаю речь...'

    try {
        const formData = new FormData()

        formData.append(
            'file',
            currentRecording,
            'saylo-recording.webm'
        )

        const response = await fetch(
            'http://127.0.0.1:8765/transcribe',
            {
                method: 'POST',
                body: formData
            }
        )

        if (!response.ok) {
            throw new Error(
                `Сервер распознавания вернул ошибку: ${response.status}`
            )
        }

        const result = await response.json()

        const transcript =
            result.text ??
            result.transcription ??
            result.transcript ??
            ''

        elements.transcriptText.textContent =
            transcript || 'Речь не распознана.'

    } catch (error) {
        console.error('Ошибка Whisper:', error)

        elements.transcriptText.textContent =
            'Не удалось выполнить распознавание. Проверь, запущен ли Whisper на 127.0.0.1:8765.'
    } finally {
        elements.transcribeButton.disabled = !currentRecording
    }
}

async function saveRecordingAsWav() {
    if (!currentRecording) {
        return
    }

    try {
        const arrayBuffer = await currentRecording.arrayBuffer()

        const audioContext = new AudioContext()

        const audioBuffer = await audioContext.decodeAudioData(
            arrayBuffer.slice(0)
        )

        const wavBlob = encodeWav(audioBuffer)

        const url = URL.createObjectURL(wavBlob)
        const link = document.createElement('a')

        link.href = url
        link.download = `saylo-${Date.now()}.wav`

        document.body.appendChild(link)
        link.click()
        link.remove()

        URL.revokeObjectURL(url)

        await audioContext.close()

    } catch (error) {
        console.error('Ошибка сохранения WAV:', error)

        elements.audioStatus.textContent =
            'Не удалось сохранить WAV. Проверь формат аудиозаписи.'
    }
}

function encodeWav(audioBuffer) {
    const channels = audioBuffer.numberOfChannels
    const sampleRate = audioBuffer.sampleRate
    const samples = audioBuffer.length

    const bytesPerSample = 2
    const blockAlign = channels * bytesPerSample
    const dataSize = samples * blockAlign

    const buffer = new ArrayBuffer(44 + dataSize)
    const view = new DataView(buffer)

    function writeString(offset, string) {
        for (let i = 0; i < string.length; i++) {
            view.setUint8(offset + i, string.charCodeAt(i))
        }
    }

    writeString(0, 'RIFF')
    view.setUint32(4, 36 + dataSize, true)
    writeString(8, 'WAVE')
    writeString(12, 'fmt ')
    view.setUint32(16, 16, true)
    view.setUint16(20, 1, true)
    view.setUint16(22, channels, true)
    view.setUint32(24, sampleRate, true)
    view.setUint32(28, sampleRate * blockAlign, true)
    view.setUint16(32, blockAlign, true)
    view.setUint16(34, 16, true)
    writeString(36, 'data')
    view.setUint32(40, dataSize, true)

    const channelData = []

    for (let channel = 0; channel < channels; channel++) {
        channelData.push(audioBuffer.getChannelData(channel))
    }

    let offset = 44

    for (let i = 0; i < samples; i++) {
        for (let channel = 0; channel < channels; channel++) {
            let sample = channelData[channel][i]

            sample = Math.max(-1, Math.min(1, sample))

            const value = sample < 0
                ? sample * 0x8000
                : sample * 0x7FFF

            view.setInt16(offset, value, true)
            offset += 2
        }
    }

    return new Blob([buffer], {
        type: 'audio/wav'
    })
}

let activeProfileId = ''

function escapeProfileText(value) {
    return String(value ?? '').replace(/[&<>"']/g, char => ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#39;'
    })[char])
}

function showProfileMessage(message, isError = false) {
    const element = document.getElementById('profileMessage')
    if (!element) return

    element.textContent = message
    element.classList.toggle('error', isError)
}

function getPhonemeEntries(data) {
    if (Array.isArray(data)) return data
    if (Array.isArray(data?.phonemes)) return data.phonemes
    if (Array.isArray(data?.sounds?.phonemes)) return data.sounds.phonemes
    if (Array.isArray(data?.data?.phonemes)) return data.data.phonemes
    return []
}

function renderPhonemeOptions() {
    const select = document.getElementById('targetPhonemeSelect')
    if (!select) return

    select.innerHTML = ''

    const placeholder = document.createElement('option')
    placeholder.value = ''
    placeholder.textContent = 'Выберите речевой звук'
    select.appendChild(placeholder)

    const groups = [
        { type: 'vowel', label: 'Гласные звуки' },
        { type: 'consonant', label: 'Согласные звуки' }
    ]

    for (const group of groups) {
        const entries = phonemeEntries.filter(item => item.type === group.type)
        if (!entries.length) continue

        const optionGroup = document.createElement('optgroup')
        optionGroup.label = group.label

        for (const phoneme of entries) {
            if (!phoneme.soundId) continue

            const option = document.createElement('option')
            option.value = phoneme.soundId

            const symbol = phoneme.phoneme || phoneme.symbol || phoneme.soundId
            const description = phoneme.description
                ? ` — ${phoneme.description}`
                : ''

            option.textContent = `${symbol}${description}`
            optionGroup.appendChild(option)
        }

        select.appendChild(optionGroup)
    }

    if (!phonemeEntries.length) {
        select.innerHTML = '<option value="">Фонемы не найдены</option>'
    }
}

async function loadPhonemeDictionary() {
    if (!window.saylo?.getPhonemeData) {
        throw new Error('Метод получения фонетического справочника недоступен.')
    }

    const data = await window.saylo.getPhonemeData()
    phonemeEntries = getPhonemeEntries(data)
    renderPhonemeOptions()

    if (!phonemeEntries.length) {
        showProfileMessage('Справочник не вернул список фонем.', true)
    }
}

async function loadTargetSounds(profileId) {
    const container = document.getElementById('targetSoundsList')
    if (!container) return

    if (!profileId) {
        container.innerHTML = '<p class="empty-progress">Сначала выберите профиль.</p>'
        return
    }

    const sounds = await window.saylo.getTargetSounds(profileId)

    if (!Array.isArray(sounds) || !sounds.length) {
        container.innerHTML = '<p class="empty-progress">Целевые звуки пока не назначены.</p>'
        return
    }

    container.innerHTML = sounds.map(sound => {
        const phoneme = phonemeEntries.find(item => item.soundId === sound.soundId)
        const symbol = phoneme?.phoneme || sound.soundId
        const stage = sound.currentStageId || 'Этап не указан'
        const status = sound.active === false ? 'Неактивен' : 'Активен'

        return `
            <div class="target-sound-item">
                <strong>${escapeProfileText(symbol)}</strong>
                <span>${escapeProfileText(stage)}</span>
                <small>${escapeProfileText(status)}</small>
            </div>
        `
    }).join('')
}

async function loadParentProfiles() {
    if (!window.saylo) {
        showProfileMessage('Связь с основным модулем Saylo недоступна.', true)
        return
    }

    try {
        const profiles = await window.saylo.getProfiles()
        const select = document.getElementById('activeProfileSelect')
        const cardsContainer = document.getElementById('savedProfilesList')

        if (!select) return

        select.innerHTML = ''

        if (!Array.isArray(profiles) || !profiles.length) {
            select.innerHTML = '<option value="">Профили пока не созданы</option>'

            if (cardsContainer) {
                cardsContainer.innerHTML =
                    '<p class="empty-progress">Сохранённых профилей пока нет.</p>'
            }

            activeProfileId = ''
            await loadTargetSounds('')
            return
        }

        for (const profile of profiles) {
            const option = document.createElement('option')
            option.value = profile.id
            option.textContent = profile.displayName || 'Без имени'
            select.appendChild(option)
        }

        const previousExists = profiles.some(
            profile => profile.id === activeProfileId
        )

        activeProfileId = previousExists
            ? activeProfileId
            : profiles[0].id

        select.value = activeProfileId

        if (cardsContainer) {
            cardsContainer.innerHTML = ''

            for (const profile of profiles) {
                const card = document.createElement('button')
                const isSelected = profile.id === activeProfileId
                const soundCount = Array.isArray(profile.targetSounds)
                    ? profile.targetSounds.length
                    : 0

                card.type = 'button'
                card.className = isSelected
                    ? 'saved-profile-item selected'
                    : 'saved-profile-item'

                const name = document.createElement('span')
                name.className = 'saved-profile-name'
                name.textContent = profile.displayName || 'Без имени'

                const details = document.createElement('span')
                details.className = 'saved-profile-details'

                const notes = profile.notes
                    ? `${profile.notes} · `
                    : ''

                details.textContent =
                    `${notes}Назначено звуков: ${soundCount}`

                card.appendChild(name)
                card.appendChild(details)

                if (isSelected) {
                    const selectedLabel = document.createElement('span')
                    selectedLabel.className = 'saved-profile-selected'
                    selectedLabel.textContent = 'Выбранный профиль'
                    card.appendChild(selectedLabel)
                }

                card.addEventListener('click', async () => {
                    activeProfileId = profile.id
                    select.value = profile.id

                    await loadParentProfiles()
                    showProfileMessage('')
                })

                cardsContainer.appendChild(card)
            }
        }

        await loadTargetSounds(activeProfileId)
    } catch (error) {
        showProfileMessage(`Ошибка загрузки профилей: ${error.message}`, true)
    }
}

            displayName,

function hideSplashScreen() {
    const splash = document.getElementById('splashScreen')

    if (!splash) return

    splash.classList.add('hidden')

    setTimeout(() => {
        splash.remove()
    }, 850)

function initializeSplashScreen() {
    const splash = document.getElementById('splashScreen')
    const image = splash?.querySelector('.splash-image')

    if (!splash || !image) return

    const showDuration = 2200

    const startTime = Date.now()

    const finishSplash = () => {
        const elapsed = Date.now() - startTime
        const remaining = Math.max(0, showDuration - elapsed)

        setTimeout(hideSplashScreen, remaining)
    }

    if (image.complete && image.naturalWidth > 0) {
        finishSplash()
    } else {
        image.addEventListener('load', finishSplash, { once: true })
        image.addEventListener('error', () => {
            console.error('Не удалось загрузить заставку Saylo:', image.src)
            finishSplash()
        }, { once: true })
    }
}

import { Assets } from 'pixi.js'
import { ASSET_MANIFEST } from './manifest.js'

// ============================================================
// SAYLO GAME ENGINE
// ASSET MANAGER
// PixiJS 8.23.0
//
// Ответственность:
// - регистрация ресурсов;
// - загрузка отдельных ресурсов;
// - загрузка тематических наборов;
// - получение загруженных ресурсов;
// - освобождение ресурсов;
// - контроль состояния загрузки.
// ============================================================

export class AssetManager {

    constructor({
        manifest = ASSET_MANIFEST,
        assetsApi = Assets
    } = {}) {

        if (!manifest || !manifest.assets || !manifest.bundles) {
            throw new TypeError(
                '[Saylo Assets] Некорректный манифест ресурсов.'
            )
        }

        if (
            !assetsApi ||
            typeof assetsApi.addBundle !== 'function' ||
            typeof assetsApi.load !== 'function' ||
            typeof assetsApi.loadBundle !== 'function' ||
            typeof assetsApi.unloadBundle !== 'function' ||
            typeof assetsApi.get !== 'function'
        ) {
            throw new TypeError(
                '[Saylo Assets] Некорректный API PixiJS Assets.'
            )
        }

        this.manifest = manifest
        this.assetsApi = assetsApi

        this.registeredBundles = new Set()
        this.loadedBundles = new Set()

        this.loadedAssets = new Map()
        this.loadingAssets = new Map()
        this.loadingBundles = new Map()

        this.destroyed = false
        this.initialized = false
    }

    // --------------------------------------------------------
    // РЕГИСТРАЦИЯ РЕСУРСОВ
    // --------------------------------------------------------

    initialize() {

        this.ensureActive()

        if (this.initialized) {
            return true
        }

        try {

            for (const [bundleName, assetKeys] of
                Object.entries(this.manifest.bundles)) {

                if (
                    !Array.isArray(assetKeys) ||
                    assetKeys.length === 0
                ) {
                    throw new Error(
                        `[Saylo Assets] Пустой набор: ${bundleName}`
                    )
                }

                const definitions = assetKeys.map(key => {

                    const asset = this.manifest.assets[key]

                    if (
                        !asset ||
                        typeof asset.alias !== 'string' ||
                        typeof asset.src !== 'string'
                    ) {
                        throw new Error(
                            `[Saylo Assets] Неизвестный ресурс: ${key}`
                        )
                    }

                    return {
                        alias: asset.alias,
                        src: asset.src
                    }
                })

                this.assetsApi.addBundle(
                    `saylo-${bundleName}`,
                    definitions
                )

                this.registeredBundles.add(bundleName)
            }

            this.initialized = true

            console.log(
                '[Saylo Assets] Каталог ресурсов зарегистрирован.',
                {
                    assets: Object.keys(this.manifest.assets).length,
                    bundles: this.registeredBundles.size
                }
            )

            return true

        } catch (error) {

            console.error(
                '[Saylo Assets] Ошибка регистрации ресурсов:',
                error
            )

            this.registeredBundles.clear()
            this.initialized = false

            throw error
        }
    }

    // --------------------------------------------------------
    // ЗАГРУЗКА ОДНОГО РЕСУРСА
    // --------------------------------------------------------

    async load(assetKey) {

        this.ensureActive()
        this.initialize()

        const definition = this.manifest.assets[assetKey]

        if (!definition) {
            throw new Error(
                `[Saylo Assets] Ресурс не найден: ${assetKey}`
            )
        }

        if (this.loadedAssets.has(assetKey)) {
            return this.loadedAssets.get(assetKey)
        }

        if (this.loadingAssets.has(assetKey)) {
            return this.loadingAssets.get(assetKey)
        }

        const promise = this.assetsApi
            .load(definition.alias)
            .then(resource => {

                this.ensureActive()

                this.loadedAssets.set(assetKey, resource)

                console.log(
                    `[Saylo Assets] Загружен: ${assetKey}`
                )

                return resource
            })
            .finally(() => {
                this.loadingAssets.delete(assetKey)
            })

        this.loadingAssets.set(assetKey, promise)

        return promise
    }

    // --------------------------------------------------------
    // ЗАГРУЗКА НАБОРА РЕСУРСОВ
    // --------------------------------------------------------

    async loadBundle(bundleName, onProgress = null) {

        this.ensureActive()
        this.initialize()

        if (!this.registeredBundles.has(bundleName)) {
            throw new Error(
                `[Saylo Assets] Набор не найден: ${bundleName}`
            )
        }

        if (typeof onProgress !== 'function' && onProgress !== null) {
            throw new TypeError(
                '[Saylo Assets] onProgress должен быть функцией.'
            )
        }

        if (this.loadedBundles.has(bundleName)) {

            const cached = {}

            for (const key of this.manifest.bundles[bundleName]) {

                const definition = this.manifest.assets[key]

                cached[definition.alias] =
                    this.loadedAssets.get(key) ??
                    this.assetsApi.get(definition.alias)
            }

            return cached
        }

        if (this.loadingBundles.has(bundleName)) {
            return this.loadingBundles.get(bundleName)
        }

        const promise = this.assetsApi
            .loadBundle(
                `saylo-${bundleName}`,
                onProgress || undefined
            )
            .then(resources => {

                this.ensureActive()

                for (const key of this.manifest.bundles[bundleName]) {

                    const definition = this.manifest.assets[key]

                    const resource =
                        resources?.[definition.alias] ??
                        this.assetsApi.get(definition.alias)

                    if (resource !== undefined) {
                        this.loadedAssets.set(key, resource)
                    }
                }

                this.loadedBundles.add(bundleName)

                console.log(
                    `[Saylo Assets] Набор загружен: ${bundleName}`
                )

                return resources
            })
            .finally(() => {
                this.loadingBundles.delete(bundleName)
            })

        this.loadingBundles.set(bundleName, promise)

        return promise
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ РЕСУРСА
    // --------------------------------------------------------

    get(assetKey) {

        this.ensureActive()

        if (!this.manifest.assets[assetKey]) {
            return null
        }

        return this.loadedAssets.get(assetKey) ?? null
    }

    has(assetKey) {

        this.ensureActive()

        return this.loadedAssets.has(assetKey)
    }

    // --------------------------------------------------------
    // ОСВОБОЖДЕНИЕ НАБОРА
    // --------------------------------------------------------

    async unloadBundle(bundleName) {

        this.ensureActive()

        if (!this.loadedBundles.has(bundleName)) {
            return false
        }

        await this.assetsApi.unloadBundle(
            `saylo-${bundleName}`
        )

        for (const key of this.manifest.bundles[bundleName]) {
            this.loadedAssets.delete(key)
        }

        this.loadedBundles.delete(bundleName)

        console.log(
            `[Saylo Assets] Набор освобождён: ${bundleName}`
        )

        return true
    }

    // --------------------------------------------------------
    // СОСТОЯНИЕ
    // --------------------------------------------------------

    getStatus() {

        return Object.freeze({

            initialized: this.initialized,
            destroyed: this.destroyed,

            registeredBundles: [
                ...this.registeredBundles
            ],

            loadedBundles: [
                ...this.loadedBundles
            ],

            loadedAssets: [
                ...this.loadedAssets.keys()
            ],

            loadingAssets: [
                ...this.loadingAssets.keys()
            ],

            loadingBundles: [
                ...this.loadingBundles.keys()
            ]

        })
    }

    // --------------------------------------------------------
    // ПРОВЕРКА АКТИВНОСТИ
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[Saylo Assets] Менеджер уничтожен.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ МЕНЕДЖЕРА
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) {
            return
        }

        // Не уничтожаем текстуры автоматически:
        // они могут использоваться другими объектами PixiJS.

        this.loadedAssets.clear()
        this.loadingAssets.clear()
        this.loadingBundles.clear()

        this.registeredBundles.clear()
        this.loadedBundles.clear()

        this.initialized = false
        this.destroyed = true

        console.log(
            '[Saylo Assets] Менеджер остановлен.'
        )
    }
}

// SAYLO GAME ENGINE
// CORE: EVENT BUS
// Версия архитектуры: 1.0
// ============================================================

export class EventBus {

    constructor() {
        this.listeners = new Map()
        this.destroyed = false
    }

    // --------------------------------------------------------
    // ПОДПИСКА НА СОБЫТИЕ
    // --------------------------------------------------------

    on(eventName, callback) {

        this.validate(eventName, callback)

        if (this.destroyed) {
            throw new Error('[EventBus] Шина событий уничтожена.')
        }

        if (!this.listeners.has(eventName)) {
            this.listeners.set(eventName, new Set())
        }

        const callbacks = this.listeners.get(eventName)

        callbacks.add(callback)

        // Возвращаем функцию отмены подписки
        return () => this.off(eventName, callback)
    }

    // --------------------------------------------------------
    // ПОДПИСКА НА ОДНОРАЗОВОЕ СОБЫТИЕ
    // --------------------------------------------------------

    once(eventName, callback) {

        this.validate(eventName, callback)

        const unsubscribe = this.on(eventName, (...args) => {
            unsubscribe()
            callback(...args)
        })

        return unsubscribe
    }

    // --------------------------------------------------------
    // ОТМЕНА ПОДПИСКИ
    // --------------------------------------------------------

    off(eventName, callback) {

        const callbacks = this.listeners.get(eventName)

        if (!callbacks) return false

        const removed = callbacks.delete(callback)

        if (callbacks.size === 0) {
            this.listeners.delete(eventName)
        }

        return removed
    }

    // --------------------------------------------------------
    // ПУБЛИКАЦИЯ СОБЫТИЯ
    // --------------------------------------------------------

    emit(eventName, ...args) {

        if (this.destroyed) return false

        const callbacks = this.listeners.get(eventName)

        if (!callbacks || callbacks.size === 0) {
            return false
        }

        // Копия защищает обход от изменения подписок
        // непосредственно во время публикации события.
        for (const callback of [...callbacks]) {

            try {
                callback(...args)
            } catch (error) {
                console.error(
                    `[EventBus] Ошибка обработчика события "${eventName}":`,
                    error
                )
            }
        }

        return true
    }

    // --------------------------------------------------------
    // УДАЛЕНИЕ ВСЕХ ПОДПИСОК НА СОБЫТИЕ
    // --------------------------------------------------------

    offAll(eventName) {

        if (eventName === undefined) {
            this.listeners.clear()
            return
        }

        this.listeners.delete(eventName)
    }

    // --------------------------------------------------------
    // ПРОВЕРКА НАЛИЧИЯ ПОДПИСЧИКОВ
    // --------------------------------------------------------

    hasListeners(eventName) {

        return (this.listeners.get(eventName)?.size ?? 0) > 0
    }

    // --------------------------------------------------------
    // КОЛИЧЕСТВО ПОДПИСЧИКОВ
    // --------------------------------------------------------

    listenerCount(eventName) {

        return this.listeners.get(eventName)?.size ?? 0
    }

    // --------------------------------------------------------
    // ПРОВЕРКА ПАРАМЕТРОВ
    // --------------------------------------------------------

    validate(eventName, callback) {

        if (typeof eventName !== 'string' || !eventName.trim()) {
            throw new TypeError(
                '[EventBus] Имя события должно быть непустой строкой.'
            )
        }

        if (typeof callback !== 'function') {
            throw new TypeError(
                '[EventBus] Обработчик события должен быть функцией.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ ШИНЫ СОБЫТИЙ
    // --------------------------------------------------------

    destroy() {

        this.listeners.clear()
        this.destroyed = true
    }
}

import 'pixi.js/unsafe-eval'

import { Application } from 'pixi.js'

import { createEngineConfig } from './EngineConfig.js'
import { EventBus } from './EventBus.js'
import { GameState } from './GameState.js'
import { GameLoop } from './GameLoop.js'

import { World } from '../world/World.js'
import { Camera } from '../world/Camera.js'

import { AssetManager } from '../assets/AssetManager.js'

import { AnimationSystem } from '../systems/AnimationSystem.js'
import { RenderingSystem } from '../systems/RenderingSystem.js'
import { HiveScene } from '../scenes/HiveScene.js'

// ============================================================
// SAYLO GAME ENGINE
// Центральное управление игровым движком
// PixiJS 8.x
// ============================================================

export class GameEngine {

    constructor(config = {}) {
        this.config = createEngineConfig(config)

        this.app = null
        this.host = null

        this.eventBus = null
        this.gameState = null
        this.gameLoop = null

        this.world = null
        this.camera = null

        this.assetManager = null
        this.renderingSystem = null
        this.scene = null

        this.initialized = false
        this.appInitialized = false
        this.initializing = false
        this.initPromise = null
    }

    // --------------------------------------------------------
    // INITIALIZATION
    // --------------------------------------------------------

    initialize() {
        if (this.initialized) {
            console.log('[Saylo Game] Движок уже запущен.')
            return Promise.resolve(true)
        }

        if (this.initPromise) {
            return this.initPromise
        }

        this.initPromise = this.initializeInternal()

        return this.initPromise.finally(() => {
            this.initPromise = null
        })
    }

    // --------------------------------------------------------
    // СОЗДАНИЕ И ЗАПУСК
    // --------------------------------------------------------

    async initializeInternal() {
        if (this.initializing) return false

        this.initializing = true

        const host = document.getElementById('hiveGameRoot')

        if (!host) {
            console.error('[Saylo Game] Контейнер hiveGameRoot не найден.')
            this.initializing = false
            return false
        }

        this.host = host

        try {
            // 1. PIXI APPLICATION

            this.app = new Application()

            const rendererConfig = this.config.renderer

            await this.app.init({
                resizeTo: host,
                backgroundColor: this.config.world.backgroundColor,
                antialias: rendererConfig.antialias,
                resolution: Math.min(
                    window.devicePixelRatio || 1,
                    this.config.performance.maxDevicePixelRatio
                ),
                autoDensity: rendererConfig.autoDensity,
                preference: rendererConfig.preference,
                powerPreference: rendererConfig.powerPreference,
                autoStart: false
            })

            this.appInitialized = true

            // Используем только собственный игровой цикл.
            this.app.ticker.stop()

            host.appendChild(this.app.canvas)

            // 2. EVENT BUS

            this.eventBus = new EventBus()
            this.animationSystem = new AnimationSystem({
                eventBus: this.eventBus
            })

            // 3. GAME STATE

            this.gameState = new GameState(this.eventBus)

            // 4. ASSET MANAGER

            this.assetManager = new AssetManager()

            this.assetManager.initialize()

            await this.assetManager.loadBundle('home')

            console.log('[Saylo Game] Ресурсы домашней сцены загружены.')

            // 5. WORLD

            this.world = new World({
                width: this.config.world.width,
                height: this.config.world.height
            })

            this.world.mount(this.app.stage)

            // 6. CAMERA

            this.camera = new Camera({
                world: this.world,
                viewportWidth: this.app.screen.width,
                viewportHeight: this.app.screen.height,
                scaleMode: this.config.world.scaleMode
            })

            // 7. RENDERING SYSTEM

            this.renderingSystem = new RenderingSystem({
                app: this.app,
                world: this.world,
                camera: this.camera,
                eventBus: this.eventBus,
                config: {
                    autoRender: false,
                    targetFPS: this.config.loop.targetFPS,
                    maxDeltaTime: this.config.loop.maxDeltaTime,
                    pauseWhenHidden: this.config.performance.pauseWhenHidden
                }
            })

            // 8. HIVE SCENE

            this.scene = new HiveScene(
                this.world,
                this.assetManager,
                {
                    animationSystem: this.animationSystem,
                    eventBus: this.eventBus
                }
            )

            await this.scene.initialize()

            // 9. GAME LOOP

            this.gameLoop = new GameLoop({
                config: this.config.loop,
                eventBus: this.eventBus,
                pauseWhenHidden: this.config.performance.pauseWhenHidden,

                onUpdate: (deltaTime, elapsedTime) => {
                    this.update(deltaTime, elapsedTime)
                }
            })

            // 10. INITIAL STATE

            this.gameState.update({
                initialized: true,
                running: true,
                paused: false,
                activeScene: 'hive',
                mode: 'home'
            })

            this.initialized = true
            this.initializing = false

            this.renderingSystem.render()
            this.gameLoop.start()

            this.eventBus.emit('engine:initialized', {
                scene: 'hive'
            })

            console.log('[Saylo Game] Игровой движок успешно запущен.')

            return true

        } catch (error) {
            console.error('[Saylo Game] Ошибка запуска движка:', error)

            this.cleanupResources()

            this.initializing = false
            this.initialized = false

            return false
        }
    }

    // --------------------------------------------------------
    // MAIN UPDATE
    // --------------------------------------------------------

    update(deltaTime, elapsedTime) {
        if (!this.initialized && !this.initializing) return false

        if (!Number.isFinite(deltaTime) || deltaTime < 0) {
            throw new TypeError('[Saylo Game] Некорректный deltaTime.')
        }

        this.animationSystem?.update(deltaTime)
        this.world?.update(deltaTime, elapsedTime)
        this.renderingSystem?.render()

        return true
    }

    // --------------------------------------------------------
    // PAUSE
    // --------------------------------------------------------

    pause() {
        if (!this.initialized || !this.gameLoop) return false

        const paused = this.gameLoop.pause()

        if (paused) {
            this.gameState.update({
                running: false,
                paused: true
            })

            this.renderingSystem?.pause()
        }

        return paused
    }

    // --------------------------------------------------------
    // RESUME
    // --------------------------------------------------------

    resume() {
        if (!this.initialized || !this.gameLoop) return false

        this.renderingSystem?.resume()

        const resumed = this.gameLoop.resume()

        if (resumed) {
            this.gameState.update({
                running: true,
                paused: false
            })
        }

        return resumed
    }

    // --------------------------------------------------------
    // STATE
    // --------------------------------------------------------

    getState() {
        return this.gameState?.getAll() ?? null
    }

    // --------------------------------------------------------
    // STATISTICS
    // --------------------------------------------------------

    getStats() {
        return {
            initialized: this.initialized,
            gameState: this.getState(),

            assets: this.assetManager
                ? this.assetManager.getStatus()
                : null,

            rendering: this.renderingSystem
                ? this.renderingSystem.getStats()
                : null
        }
    }

    // --------------------------------------------------------
    // CLEANUP
    // --------------------------------------------------------

    cleanupResources() {
        const cleanup = (name, callback) => {
            try {
                callback()
            } catch (error) {
                console.error(`[Saylo Game] Ошибка очистки ${name}:`, error)
            }
        }

        // 1. Останавливаем игровой цикл.

        if (this.gameLoop) {
            cleanup('GameLoop', () => this.gameLoop.destroy())
            this.gameLoop = null
        }

        // 2. Уничтожаем сцену.

        if (this.scene) {
            cleanup('HiveScene', () => this.scene.destroy())
            this.scene = null
        }

        // 3. Останавливаем систему анимаций.

        if (this.animationSystem) {
            cleanup('AnimationSystem', () => this.animationSystem.destroy())
            this.animationSystem = null
        }

        // 4. Останавливаем менеджер ресурсов.
        // Текстуры PixiJS автоматически не выгружаем.

        if (this.assetManager) {
            cleanup('AssetManager', () => this.assetManager.destroy())
            this.assetManager = null
        }

        // 4. Rendering.

        if (this.renderingSystem) {
            cleanup('RenderingSystem', () => this.renderingSystem.destroy())
            this.renderingSystem = null
        }

        // 5. Camera.

        if (this.camera) {
            cleanup('Camera', () => this.camera.destroy())
            this.camera = null
        }

        // 6. World.

        if (this.world) {
            cleanup('World', () => this.world.destroy())
            this.world = null
        }

        // 7. State.

        if (this.gameState) {
            cleanup('GameState', () => this.gameState.destroy())
            this.gameState = null
        }

        // 8. Event bus.

        if (this.eventBus) {
            cleanup('EventBus', () => this.eventBus.destroy())
            this.eventBus = null
        }

        // 9. Canvas.

        if (this.app?.canvas?.parentNode === this.host) {
            this.app.canvas.remove()
        }

        // 10. Pixi application.

        if (this.app) {
            cleanup('Pixi Application', () => this.app.destroy())
            this.app = null
        }

        this.host = null
        this.appInitialized = false
        this.initialized = false
    }

    // --------------------------------------------------------
    // DESTROY
    // --------------------------------------------------------

    destroy() {
        if (
            !this.app &&
            !this.world &&
            !this.gameLoop &&
            !this.assetManager &&
            !this.initialized
        ) {
            return
        }

        this.cleanupResources()
        this.initializing = false

        console.log('[Saylo Game] Движок остановлен.')
    }
}

// ============================================================
// SAYLO GAME ENGINE
// CORE: GAME LOOP
// Версия архитектуры: 1.0
// ============================================================

export class GameLoop {

    constructor({
        onUpdate,
        config = {},
        eventBus = null,
        pauseWhenHidden = true
    } = {}) {

        if (typeof onUpdate !== 'function') {
            throw new TypeError(
                '[GameLoop] Необходимо передать функцию onUpdate.'
            )
        }

        this.onUpdate = onUpdate
        this.eventBus = eventBus

        this.fixedTimeStep = config.fixedTimeStep ?? (1 / 60)
        this.maxDeltaTime = config.maxDeltaTime ?? 0.05
        this.maxSubSteps = 5
        this.pauseWhenHidden = pauseWhenHidden

        if (
            !Number.isFinite(this.fixedTimeStep) ||
            this.fixedTimeStep <= 0
        ) {
            throw new RangeError(
                '[GameLoop] fixedTimeStep должен быть положительным числом.'
            )
        }

        if (
            !Number.isFinite(this.maxDeltaTime) ||
            this.maxDeltaTime <= 0
        ) {
            throw new RangeError(
                '[GameLoop] maxDeltaTime должен быть положительным числом.'
            )
        }

        this.running = false
        this.paused = false
        this.destroyed = false

        this.frameId = null
        this.lastTimestamp = null

        this.accumulator = 0
        this.elapsedTime = 0

        this.wasRunningBeforeHidden = false

        this.handleVisibilityChange =
            this.handleVisibilityChange.bind(this)

        if (this.pauseWhenHidden && typeof document !== 'undefined') {
            document.addEventListener(
                'visibilitychange',
                this.handleVisibilityChange
            )
        }
    }

    // --------------------------------------------------------
    // ЗАПУСК
    // --------------------------------------------------------

    start() {

        this.ensureActive()

        if (this.running) return false

        if (typeof requestAnimationFrame !== 'function') {
            throw new Error(
                '[GameLoop] requestAnimationFrame недоступен.'
            )
        }

        this.running = true
        this.paused = false

        this.lastTimestamp = null
        this.accumulator = 0

        this.scheduleNextFrame()

        this.eventBus?.emit('loop:start')

        return true
    }

    // --------------------------------------------------------
    // ПАУЗА
    // --------------------------------------------------------

    pause() {

        this.ensureActive()

        if (!this.running) return false

        this.running = false
        this.paused = true

        this.cancelFrame()

        this.lastTimestamp = null
        this.accumulator = 0

        this.eventBus?.emit('loop:pause')

        return true
    }

    // --------------------------------------------------------
    // ВОЗОБНОВЛЕНИЕ
    // --------------------------------------------------------

    resume() {

        this.ensureActive()

        if (!this.paused) return false

        this.paused = false
        this.running = true

        this.lastTimestamp = null
        this.accumulator = 0

        this.scheduleNextFrame()

        this.eventBus?.emit('loop:resume')

        return true
    }

    // --------------------------------------------------------
    // ОСТАНОВКА
    // --------------------------------------------------------

    stop() {

        this.ensureActive()

        if (!this.running && !this.paused) return false

        this.running = false
        this.paused = false

        this.cancelFrame()

        this.lastTimestamp = null
        this.accumulator = 0
        this.elapsedTime = 0

        this.eventBus?.emit('loop:stop')

        return true
    }

    // --------------------------------------------------------
    // ПЛАНИРОВАНИЕ КАДРА
    // --------------------------------------------------------

    scheduleNextFrame() {

        if (!this.running || this.destroyed) return

        this.frameId = requestAnimationFrame(
            this.handleFrame.bind(this)
        )
    }

    cancelFrame() {

        if (this.frameId === null) return

        cancelAnimationFrame(this.frameId)
        this.frameId = null
    }

    // --------------------------------------------------------
    // ОБРАБОТКА КАДРА
    // --------------------------------------------------------

    handleFrame(timestamp) {

        this.frameId = null

        if (!this.running || this.destroyed) return

        if (this.lastTimestamp === null) {
            this.lastTimestamp = timestamp
            this.scheduleNextFrame()
            return
        }

        const rawDelta = Math.max(
            0,
            (timestamp - this.lastTimestamp) / 1000
        )

        this.lastTimestamp = timestamp

        const deltaTime = Math.min(
            rawDelta,
            this.maxDeltaTime
        )

        this.accumulator += deltaTime

        let steps = 0

        try {

            while (
                this.accumulator >= this.fixedTimeStep &&
                steps < this.maxSubSteps
            ) {

                this.onUpdate(
                    this.fixedTimeStep,
                    this.elapsedTime
                )

                this.elapsedTime += this.fixedTimeStep
                this.accumulator -= this.fixedTimeStep

                steps++
            }

            // Не допускаем накопления большого временного долга.

            if (
                steps === this.maxSubSteps &&
                this.accumulator >= this.fixedTimeStep
            ) {
                this.accumulator %= this.fixedTimeStep
            }

        } catch (error) {

            console.error(
                '[GameLoop] Ошибка обновления игрового цикла:',
                error
            )

            this.stop()

            this.eventBus?.emit('loop:error', {
                error
            })

            return
        }

        this.scheduleNextFrame()
    }

    // --------------------------------------------------------
    // ВИДИМОСТЬ ОКНА
    // --------------------------------------------------------

    handleVisibilityChange() {

        if (!this.pauseWhenHidden || this.destroyed) return

        if (document.hidden) {

            this.wasRunningBeforeHidden = this.running

            if (this.running) {
                this.pause()
            }

        } else {

            if (this.wasRunningBeforeHidden && this.paused) {
                this.resume()
            }

            this.wasRunningBeforeHidden = false
        }
    }

    // --------------------------------------------------------
    // ПРОВЕРКА ДОСТУПНОСТИ
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[GameLoop] Игровой цикл уничтожен.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.cancelFrame()

        if (this.pauseWhenHidden && typeof document !== 'undefined') {
            document.removeEventListener(
                'visibilitychange',
                this.handleVisibilityChange
            )
        }

        this.running = false
        this.paused = false

        this.onUpdate = null
        this.eventBus = null

        this.lastTimestamp = null
        this.accumulator = 0
        this.elapsedTime = 0

        this.destroyed = true
    }
}

export class GameState {

    constructor(eventBus = null) {

        this.eventBus = eventBus

        this.state = {
            initialized: false,
            running: false,
            paused: false,
            activeScene: null,
            selectedEntity: null,
            mode: 'home'
        }

        this.destroyed = false
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ СОСТОЯНИЯ
    // --------------------------------------------------------

    get(key) {

        this.ensureActive()

        if (!Object.hasOwn(this.state, key)) {
            return undefined
        }

        return this.state[key]
    }

    getAll() {

        this.ensureActive()

        return Object.freeze({
            ...this.state
        })
    }

    // --------------------------------------------------------
    // ИЗМЕНЕНИЕ СОСТОЯНИЯ
    // --------------------------------------------------------

    set(key, value) {

        this.ensureActive()

        if (!Object.hasOwn(this.state, key)) {
            throw new Error(
                `[GameState] Неизвестный параметр состояния: ${key}`
            )
        }

        const previousValue = this.state[key]

        if (Object.is(previousValue, value)) {
            return false
        }

        this.state[key] = value

        this.eventBus?.emit('state:changed', {
            key,
            value,
            previousValue
        })

        this.eventBus?.emit(`state:${key}`, {
            value,
            previousValue
        })

        return true
    }

    // --------------------------------------------------------
    // МАССОВОЕ ИЗМЕНЕНИЕ СОСТОЯНИЯ
    // --------------------------------------------------------

    update(values = {}) {

        this.ensureActive()

        if (!values || typeof values !== 'object' || Array.isArray(values)) {
            throw new TypeError(
                '[GameState] Ожидается объект с параметрами состояния.'
            )
        }

        const entries = Object.entries(values)

        // Сначала проверяем все ключи.
        // При ошибке состояние останется неизменным.

        for (const [key] of entries) {

            if (!Object.hasOwn(this.state, key)) {
                throw new Error(
                    `[GameState] Неизвестный параметр состояния: ${key}`
                )
            }
        }

        // Только после успешной проверки изменяем состояние.

        const changes = []

        for (const [key, value] of entries) {

            const previousValue = this.state[key]

            if (Object.is(previousValue, value)) {
                continue
            }

            this.state[key] = value

            changes.push({
                key,
                value,
                previousValue
            })
        }

        // Уведомляем подписчиков об изменениях.

        for (const change of changes) {

            this.eventBus?.emit('state:changed', change)

            this.eventBus?.emit(`state:${change.key}`, {
                value: change.value,
                previousValue: change.previousValue
            })
        }

        return changes.length
    }

    // --------------------------------------------------------
    // СБРОС СОСТОЯНИЯ
    // --------------------------------------------------------

    reset() {

        this.ensureActive()

        const previousState = this.getAll()

        this.state = {
            initialized: false,
            running: false,
            paused: false,
            activeScene: null,
            selectedEntity: null,
            mode: 'home'
        }

        this.eventBus?.emit('state:reset', {
            previousState,
            state: this.getAll()
        })
    }

    // --------------------------------------------------------
    // ПРОВЕРКА ДОСТУПНОСТИ
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[GameState] Хранилище состояния уничтожено.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.state = null
        this.eventBus = null
        this.destroyed = true
    }
}

// BEE ANIMATOR
// Переключение ракурсов с плавным переходом
// ============================================================

export class BeeAnimator {

    constructor({
        character,
        animationSystem,
        textures,
        initialPose = 'front_standing'
    } = {}) {

        if (!character?.container || !character.sprite) {
            throw new TypeError('[BeeAnimator] Персонаж не подключён.')
        }

        if (
            !animationSystem ||
            typeof animationSystem.animate !== 'function' ||
            typeof animationSystem.cancelTarget !== 'function'
        ) {
            throw new TypeError('[BeeAnimator] AnimationSystem не подключён.')
        }

        if (!textures || !textures[initialPose]) {
            throw new TypeError('[BeeAnimator] Не найдена начальная поза.')
        }

        this.character = character
        this.animationSystem = animationSystem
        this.textures = textures

        this.currentPose = initialPose
        this.transition = null

        this.destroyed = false
    }

    // --------------------------------------------------------
    // ПЕРЕКЛЮЧЕНИЕ РАКУРСА
    // --------------------------------------------------------

    transitionTo(pose, {
        duration = 0.22,
        easing = 'easeInOut'
    } = {}) {

        this.ensureActive()

        const texture = this.textures[pose]

        if (!texture) {
            throw new Error(`[BeeAnimator] Неизвестная поза: ${pose}`)
        }

        if (!Number.isFinite(duration) || duration <= 0) {
            throw new RangeError('[BeeAnimator] Некорректная длительность.')
        }

        if (!this.transition && this.currentPose === pose) {
            return false
        }

        if (this.transition?.toPose === pose) {
            return false
        }

        this.cancelTransition()

        const container = this.character.container
        const from = this.character.sprite

        const to = new Sprite(texture)

        to.label = `BeePose_${pose}`
        to.anchor.set(0.5)
        to.position.set(0, 0)
        to.scale.set(
            this.character.baseSpriteScale *
            this.character.visualScale
        )
        to.alpha = 0

        container.addChild(to)

        const transition = {
            from,
            to,
            fromPose: this.currentPose,
            toPose: pose
        }

        this.transition = transition

        // Сначала затухает предыдущая поза.
        this.animationSystem.animate(
            from,
            { alpha: 0 },
            { duration, easing }
        )

        // Затем проявляется новая.
        return this.animationSystem.animate(
            to,
            { alpha: 1 },
            {
                duration,
                easing,
                onComplete: () => {
                    this.finishTransition(transition)
                }
            }
        )
    }

    // --------------------------------------------------------
    // ЗАВЕРШЕНИЕ ПЕРЕХОДА
    // --------------------------------------------------------

    finishTransition(transition) {

        if (this.transition !== transition || this.destroyed) {
            return
        }

        const { from, to, toPose } = transition

        this.transition = null

        if (from.parent === this.character.container) {
            this.character.container.removeChild(from)
        }

        from.destroy({
            texture: false,
            textureSource: false
        })

        this.character.sprite = to
        this.character.texture = to.texture

        this.currentPose = toPose

    }

    // --------------------------------------------------------
    // ОТМЕНА ПЕРЕХОДА
    // --------------------------------------------------------

    cancelTransition() {

        const transition = this.transition

        if (!transition) return false

        this.transition = null

        this.animationSystem.cancelTarget(
            transition.from,
            'bee-pose-transition-replaced'
        )

        this.animationSystem.cancelTarget(
            transition.to,
            'bee-pose-transition-replaced'
        )

        const keepNew = transition.to.alpha >= transition.from.alpha

        const keep = keepNew ? transition.to : transition.from
        const discard = keepNew ? transition.from : transition.to

        if (discard.parent === this.character.container) {
            this.character.container.removeChild(discard)
        }

        discard.destroy({
            texture: false,
            textureSource: false
        })

        keep.alpha = 1

        this.character.sprite = keep
        this.character.texture = keep.texture

        this.currentPose = keepNew
            ? transition.toPose
            : transition.fromPose

        return true
    }

    // --------------------------------------------------------
    // ИЗМЕНЕНИЕ МАСШТАБА
    // --------------------------------------------------------

    setScale(scale) {

        this.ensureActive()

        if (!Number.isFinite(scale) || scale <= 0) {
            throw new RangeError('[BeeAnimator] Некорректный масштаб.')
        }

        const sprites = new Set([
            this.character.sprite,
            this.transition?.from,
            this.transition?.to
        ])

        for (const sprite of sprites) {
            if (sprite && !sprite.destroyed) {
                sprite.scale.set(scale)
            }
        }
    }

    getCurrentPose() {
        this.ensureActive()
        return this.currentPose
    }

    ensureActive() {
        if (this.destroyed) {
            throw new Error('[BeeAnimator] Аниматор уничтожен.')
        }
    }

    destroy() {

        if (this.destroyed) return

        this.cancelTransition()

        this.character = null
        this.animationSystem = null
        this.textures = null

        this.destroyed = true
    }
}

                this.baseSpriteScale * multiplier
            )
        } else if (this.sprite) {
            this.sprite.scale.set(
                this.baseSpriteScale * multiplier
            )
        }

        return true
    }

    restoreVisualScale() {
        return this.setVisualScale(1)
    }

    // --------------------------------------------------------
    // ПОЛЁТ ПО ДУГЕ
    // --------------------------------------------------------

    flyTo(x, y, {
        duration = 1.35,
        lift = 130,
        flightState = null,
        arrivalState = 'hovering'
    } = {}) {

        this.ensureActive()

        if (
            !Number.isFinite(x) ||
            !Number.isFinite(y) ||
            !Number.isFinite(duration) ||
            duration <= 0 ||
            !Number.isFinite(lift) ||
            lift < 0
        ) {
            throw new TypeError('[Bee] Некорректные параметры полёта.')
        }

        // Прерываем спокойное покачивание и предыдущий полёт.

        this.animationSystem.cancelTarget(
            this,
            'bee-flight-started'
        )

        this.idleAnimationId = null

        if (this.flightProxy) {
            this.animationSystem.cancelTarget(
                this.flightProxy,
                'bee-flight-replaced'
            )
        }

        const start = this.position

        const selectedFlightState = flightState || (
            x < start.x - 20
                ? 'flying_left'
                : x > start.x + 20
                    ? 'flying_right'
                    : 'hovering'
        )

        this.setState(selectedFlightState, {
            duration: 0.18
        })

        const proxy = {
            id: `${this.id}-flight`,
            progress: 0
        }

        this.flightProxy = proxy

        const controlX = (start.x + x) / 2
        const controlY = Math.min(start.y, y) - lift

        this.flightAnimationId = this.animationSystem.animate(
            proxy,
            {
                progress: 1
            },
            {
                duration,
                easing: 'easeInOutCubic',

                onUpdate: progress => {

                    if (
                        this.destroyed ||
                        this.flightProxy !== proxy
                    ) {
                        return
                    }

                    // Плавная квадратичная кривая Безье.

                    const t = Math.max(0, Math.min(1, progress))
                    const inverse = 1 - t

                    const eased = t * t * (3 - 2 * t)

                    const px =
                        inverse * inverse * start.x +
                        2 * inverse * t * controlX +
                        t * t * x

                    const py =
                        inverse * inverse * start.y +
                        2 * inverse * t * controlY +
                        t * t * y

                    this.setPosition(px, py)
                },

                onComplete: () => {

                    if (
                        this.destroyed ||
                        this.flightProxy !== proxy
                    ) {
                        return
                    }

                    this.setPosition(x, y)

                    if (arrivalState) {
                        this.setState(arrivalState, {
                            duration: 0.24
                        })
                    }

                    this.flightProxy = null
                    this.flightAnimationId = null

                    this.startIdleAnimation()

                    console.log('[Saylo Bee] Перелёт завершён.')
                }
            }
        )

        console.log('[Saylo Bee] Шмелёнок отправился в полёт.')

        return this.flightAnimationId
    }

    // --------------------------------------------------------
    // ВОЗВРАЩЕНИЕ ДОМОЙ
    // --------------------------------------------------------

    flyHome(options = {}) {

        this.ensureActive()

        this.restoreVisualScale()

        return this.flyTo(
            this.homePosition.x,
            this.homePosition.y,
            {
                ...options,

                // Направление определяется фактическим маршрутом.
                // Если летим вправо — правый профиль,
                // если влево — левый профиль.

                flightState: options.flightState ?? null,
                arrivalState: options.arrivalState ?? 'idle'
            }
        )
    }

    // --------------------------------------------------------
    // ПЕРЕКЛЮЧЕНИЕ РАКУРСА
    // --------------------------------------------------------

    setPose(pose, options = {}) {

        this.ensureActive()

        if (!this.poseAnimator) return false

        return this.poseAnimator.transitionTo(pose, options)
    }

    // --------------------------------------------------------
    // ПЕРЕКЛЮЧЕНИЕ СОСТОЯНИЯ
    // --------------------------------------------------------

    setState(state, options = {}) {

        this.ensureActive()

        if (!this.stateMachine) return false

        return this.stateMachine.setState(state, options)
    }

    getPose() {
        return this.poseAnimator?.getCurrentPose() ?? null
    }

    // --------------------------------------------------------
    // ПЕРВОЕ СОСТОЯНИЕ: СПОКОЙНЫЙ ПОЛЁТ
    // --------------------------------------------------------

    startIdleAnimation() {

        this.ensureActive()

        if (this.idleAnimationId) {
            return this.idleAnimationId
        }

        const currentY = this.position.y

        this.idleAnimationId = this.animationSystem.animate(
            this,
            {
                y: currentY - this.config.idle.floatDistance
            },
            {
                duration: this.config.idle.duration,
                easing: this.config.idle.easing,
                repeat: Infinity,
                yoyo: true
            }
        )

        console.log('[Saylo Bee] Анимация спокойного полёта запущена.')

        return this.idleAnimationId
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        if (
            this.animationSystem &&
            !this.animationSystem.destroyed
        ) {
            this.animationSystem.cancelTarget(
                this,
                'bee-destroyed'
            )
        }

        this.stateMachine?.destroy()
        this.stateMachine = null

        this.poseAnimator?.destroy()
        this.poseAnimator = null

        if (this.container && !this.container.destroyed) {
            this.container.destroy({
                children: true
            })
        }

        this.container = null
        this.sprite = null
        this.texture = null
        this.poseTextures = null
        this.animationSystem = null
        this.config = null
        this.homePosition = null

        super.destroy()

        console.log('[Saylo Bee] Шмелёнок уничтожен.')
    }

}

            throw new TypeError('[BeeStateMachine] Персонаж не подключён.')
        }

        this.character = character
        this.currentState = 'idle'
        this.destroyed = false
    }

        this.ensureActive()

        const pose = BEE_STATES[state]

        if (!pose) {
            throw new Error(`[BeeStateMachine] Неизвестное состояние: ${state}`)
        }

        if (this.currentState === state) {
            return false
        }

        this.currentState = state

            'bee:state-changed',
            {
                state,
                previous,
                pose
            }
        )

        return result
    }

    getState() {
        this.ensureActive()
        return this.currentState
    }

        if (this.destroyed) {
            throw new Error('[BeeStateMachine] Машина состояний уничтожена.')
        }
    }

        if (this.destroyed) return

        this.character = null
        this.currentState = null
        this.destroyed = true
    }

/**
 * SAYLO GAME ENGINE
 * MODULE: MovementController
 *
 * Управление перемещением игровых сущностей
 *
 * Ответственность:
 * - построение маршрута через PathFinder;
 * - связь NavigationGraph с физическими Waypoints;
 * - перемещение Entity по маршруту;
 * - определение прибытия в точку;
 * - публикация событий движения.
 *
 * ВАЖНО:
 * NavigationGraph использует ID узлов графа.
 * Waypoints использует ID физических точек мира.
 * Это разные идентификаторы, и они не взаимозаменяемы.
 */

export class MovementController {
    constructor({
        entity,
        pathFinder,
        graph,
        waypoints,
        eventBus = null,
        speed = 180,
        arrivalThreshold = 3
    } = {}) {
        this.validateDependencies({
            entity,
            pathFinder,
            graph,
            waypoints,
            eventBus
        })

        this.validatePositiveNumber(speed, 'speed')
        this.validatePositiveNumber(arrivalThreshold, 'arrivalThreshold')

        this.entity = entity
        this.pathFinder = pathFinder
        this.graph = graph
        this.waypoints = waypoints
        this.eventBus = eventBus

        this.speed = speed
        this.arrivalThreshold = arrivalThreshold

        this.path = []
        this.pathIndex = 0

        this.targetNodeId = null
        this.currentNodeId = null

        this.moving = false
        this.paused = false
        this.destroyed = false

        this.distanceTravelled = 0
        this.elapsedTime = 0
    }

    // =========================================================
    // DEPENDENCIES
    // =========================================================

    validateDependencies({
        entity,
        pathFinder,
        graph,
        waypoints,
        eventBus
    }) {
        if (
            !entity ||
            typeof entity.setPosition !== 'function' ||
            typeof entity.setWaypointId !== 'function' ||
            !entity.position
        ) {
            throw new TypeError(
                'MovementController: требуется корректный Entity.'
            )
        }

        if (
            !pathFinder ||
            typeof pathFinder.findPath !== 'function'
        ) {
            throw new TypeError(
                'MovementController: требуется PathFinder.'
            )
        }

        if (
            !graph ||
            typeof graph.getNode !== 'function' ||
            typeof graph.getNodes !== 'function'
        ) {
            throw new TypeError(
                'MovementController: требуется NavigationGraph.'
            )
        }

        if (
            !waypoints ||
            typeof waypoints.get !== 'function'
        ) {
            throw new TypeError(
                'MovementController: требуется Waypoints.'
            )
        }

        if (
            eventBus !== null &&
            typeof eventBus.emit !== 'function'
        ) {
            throw new TypeError(
                'MovementController: некорректный EventBus.'
            )
        }
    }

    validatePositiveNumber(value, name) {
        if (
            typeof value !== 'number' ||
            !Number.isFinite(value) ||
            value <= 0
        ) {
            throw new TypeError(
                `MovementController: ${name} должен быть положительным числом.`
            )
        }
    }

    ensureActive() {
        if (this.destroyed) {
            throw new Error(
                'MovementController: контроллер уничтожен.'
            )
        }
    }

    // =========================================================
    // EVENTS
    // =========================================================

    emit(eventName, payload = {}) {
        if (!this.eventBus) return

        try {
            this.eventBus.emit(eventName, {
                entityId: this.entity.id ?? null,
                ...payload
            })
        } catch (error) {
            console.error(
                `[MovementController] Ошибка события ${eventName}:`,
                error
            )
        }
    }

    // =========================================================
    // WAYPOINT RESOLUTION
    // =========================================================

    getWaypointPosition(waypointId) {
        if (typeof waypointId !== 'string' || !waypointId.trim()) {
            return null
        }

        const waypoint = this.waypoints.get(waypointId)

        if (!waypoint) {
            return null
        }

        const position = waypoint.position ?? waypoint

        if (
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y)
        ) {
            return null
        }

        return {
            x: position.x,
            y: position.y
        }
    }

    getGraphNodePosition(node) {
        if (!node || typeof node.waypointId !== 'string') {
            return null
        }

        return this.getWaypointPosition(node.waypointId)
    }

    // =========================================================
    // FIND CURRENT GRAPH NODE
    // =========================================================

    findCurrentNode() {
        const physicalWaypointId = this.entity.waypointId

        // Сначала пытаемся восстановить графовый узел
        // по физической точке, к которой привязана Entity.
        if (physicalWaypointId) {
            const nodes = this.graph.getNodes()

            const matchingNode = nodes.find(
                node => node.waypointId === physicalWaypointId
            )

            if (matchingNode) {
                return matchingNode
            }
        }

        // Если привязки нет, ищем ближайший узел графа
        // относительно текущих координат сущности.
        const entityPosition = this.entity.position

        let nearestNode = null
        let nearestDistance = Infinity

        for (const node of this.graph.getNodes()) {
            const position = this.getGraphNodePosition(node)

            if (!position) continue

            const distance = this.distance(
                entityPosition,
                position
            )

            if (distance < nearestDistance) {
                nearestDistance = distance
                nearestNode = node
            }
        }

        if (!nearestNode) {
            return null
        }

        return nearestNode
    }

    // =========================================================
    // DISTANCE
    // =========================================================

    distance(a, b) {
        const dx = b.x - a.x
        const dy = b.y - a.y

        return Math.sqrt(dx * dx + dy * dy)
    }

    // =========================================================
    // MOVE TO
    // =========================================================

    moveTo(targetNodeId, options = {}) {
        this.ensureActive()

        if (
            typeof targetNodeId !== 'string' ||
            !targetNodeId.trim()
        ) {
            throw new TypeError(
                'MovementController: targetNodeId должен быть строкой.'
            )
        }

        const targetNode = this.graph.getNode(targetNodeId)

        if (!targetNode) {
            this.emit('movement:failed', {
                targetNodeId,
                reason: 'target-node-not-found'
            })

            return false
        }

        if (!this.getGraphNodePosition(targetNode)) {
            this.emit('movement:failed', {
                targetNodeId,
                reason: 'target-waypoint-not-found'
            })

            return false
        }

        const startNode = this.findCurrentNode()

        if (!startNode) {
            this.emit('movement:failed', {
                targetNodeId,
                reason: 'current-node-not-found'
            })

            return false
        }

        const result = this.pathFinder.findPath(
            startNode.id,
            targetNodeId,
            options
        )

        if (
            !result ||
            !Array.isArray(result.path) ||
            result.path.length === 0
        ) {
            this.emit('movement:failed', {
                targetNodeId,
                startNodeId: startNode.id,
                reason: 'path-not-found'
            })

            return false
        }

        // Проверяем весь маршрут до изменения текущего движения.
        const resolvedPath = []

        for (const nodeId of result.path) {
            const node = this.graph.getNode(nodeId)

            if (!node) {
                this.emit('movement:failed', {
                    targetNodeId,
                    reason: 'route-contains-unknown-node',
                    invalidNodeId: nodeId
                })

                return false
            }

            const position = this.getGraphNodePosition(node)

            if (!position) {
                this.emit('movement:failed', {
                    targetNodeId,
                    reason: 'route-contains-invalid-waypoint',
                    invalidNodeId: nodeId
                })

                return false
            }

            resolvedPath.push({
                nodeId: node.id,
                waypointId: node.waypointId,
                zoneId: node.zoneId ?? null,
                position
            })
        }

        // Если маршрут найден, прекращаем предыдущее движение.
        this.stop('replaced')

        this.path = resolvedPath
        this.pathIndex = 0

        this.currentNodeId = startNode.id
        this.targetNodeId = targetNodeId

        this.distanceTravelled = 0
        this.elapsedTime = 0

        // Начальная точка маршрута обычно совпадает
        // с текущим положением Entity.
        const firstPoint = this.path[0]

        if (
            this.distance(
                this.entity.position,
                firstPoint.position
            ) <= this.arrivalThreshold
        ) {
            this.pathIndex = 1
        }

        if (this.pathIndex >= this.path.length) {
            this.completeMovement()
            return true
        }

        this.moving = true
        this.paused = false

        this.emit('movement:started', {
            startNodeId: startNode.id,
            targetNodeId,
            route: this.path.map(point => point.nodeId)
        })

        return true
    }

    // =========================================================
    // UPDATE
    // =========================================================

    update(deltaTime) {
        this.ensureActive()

        if (!this.moving || this.paused) {
            return
        }

        if (
            typeof deltaTime !== 'number' ||
            !Number.isFinite(deltaTime) ||
            deltaTime <= 0
        ) {
            return
        }

        // Защита от резких скачков времени кадра.
        const dt = Math.min(deltaTime, 0.05)

        this.elapsedTime += dt

        let remainingDistance = this.speed * dt

        while (
            remainingDistance > 0 &&
            this.moving &&
            this.pathIndex < this.path.length
        ) {
            const target = this.path[this.pathIndex]
            const current = this.entity.position

            const distance = this.distance(current, target.position)

            if (distance <= this.arrivalThreshold) {
                this.arriveAtPoint(target)
                continue
            }

            const step = Math.min(
                remainingDistance,
                distance
            )

            const ratio = step / distance

            const nextPosition = {
                x: current.x +
                    (target.position.x - current.x) * ratio,

                y: current.y +
                    (target.position.y - current.y) * ratio
            }

            this.entity.setPosition(
                nextPosition.x,
                nextPosition.y
            )

            this.distanceTravelled += step
            remainingDistance -= step

            const remainingToTarget = this.distance(
                nextPosition,
                target.position
            )

            if (remainingToTarget <= this.arrivalThreshold) {
                this.entity.setPosition(
                    target.position.x,
                    target.position.y
                )

                this.arriveAtPoint(target)
            }
        }

        if (
            this.moving &&
            this.pathIndex >= this.path.length
        ) {
            this.completeMovement()
        }
    }

    // =========================================================
    // ARRIVAL
    // =========================================================

    arriveAtPoint(point) {
        this.entity.setWaypointId(point.waypointId)

        if (
            point.zoneId !== null &&
            typeof this.entity.setZoneId === 'function'
        ) {
            this.entity.setZoneId(point.zoneId)
        }

        this.currentNodeId = point.nodeId

        this.emit('movement:waypoint-reached', {
            nodeId: point.nodeId,
            waypointId: point.waypointId,
            zoneId: point.zoneId
        })

        this.pathIndex++
    }

    // =========================================================
    // COMPLETE
    // =========================================================

    completeMovement() {
        if (!this.moving && !this.targetNodeId) {
            return
        }

        const completedTarget = this.targetNodeId

        this.moving = false
        this.paused = false

        this.path = []
        this.pathIndex = 0

        this.targetNodeId = null

        this.emit('movement:arrived', {
            nodeId: completedTarget,
            distanceTravelled: this.distanceTravelled,
            elapsedTime: this.elapsedTime
        })
    }

    // =========================================================
    // STOP
    // =========================================================

    stop(reason = 'manual') {
        this.ensureActive()

        if (!this.moving && !this.targetNodeId) {
            return false
        }

        const previousTarget = this.targetNodeId

        this.moving = false
        this.paused = false

        this.path = []
        this.pathIndex = 0

        this.targetNodeId = null

        this.emit('movement:stopped', {
            targetNodeId: previousTarget,
            reason
        })

        return true
    }

    // =========================================================
    // PAUSE / RESUME
    // =========================================================

    pause() {
        this.ensureActive()

        if (!this.moving || this.paused) {
            return false
        }

        this.paused = true

        this.emit('movement:paused', {
            targetNodeId: this.targetNodeId
        })

        return true
    }

    resume() {
        this.ensureActive()

        if (!this.moving || !this.paused) {
            return false
        }

        this.paused = false

        this.emit('movement:resumed', {
            targetNodeId: this.targetNodeId
        })

        return true
    }

    // =========================================================
    // CONFIGURATION
    // =========================================================

    setSpeed(speed) {
        this.ensureActive()
        this.validatePositiveNumber(speed, 'speed')

        this.speed = speed
        return this.speed
    }

    setArrivalThreshold(value) {
        this.ensureActive()
        this.validatePositiveNumber(value, 'arrivalThreshold')

        this.arrivalThreshold = value
        return this.arrivalThreshold
    }

    // =========================================================
    // STATE
    // =========================================================

    isMoving() {
        return this.moving && !this.paused
    }

    isPaused() {
        return this.paused
    }

    getState() {
        return Object.freeze({
            moving: this.moving,
            paused: this.paused,
            currentNodeId: this.currentNodeId,
            targetNodeId: this.targetNodeId,
            pathIndex: this.pathIndex,
            pathLength: this.path.length,
            speed: this.speed,
            arrivalThreshold: this.arrivalThreshold,
            distanceTravelled: this.distanceTravelled,
            elapsedTime: this.elapsedTime
        })
    }

    // =========================================================
    // DESTROY
    // =========================================================

    destroy() {
        if (this.destroyed) return

        if (this.moving || this.targetNodeId) {
            this.stop('destroyed')
        }

        this.path = []

        this.entity = null
        this.pathFinder = null
        this.graph = null
        this.waypoints = null
        this.eventBus = null

        this.currentNodeId = null
        this.targetNodeId = null

        this.moving = false
        this.paused = false
        this.destroyed = true
    }
}

export class NavigationGraph {

    constructor(eventBus = null) {

        this.eventBus = eventBus

        this.nodes = new Map()
        this.edges = new Map()

        this.destroyed = false
    }

    // --------------------------------------------------------
    // ДОБАВЛЕНИЕ УЗЛА
    // --------------------------------------------------------

    addNode(id, data = {}) {

        this.ensureActive()
        this.validateId(id)

        if (
            !data ||
            typeof data !== 'object' ||
            Array.isArray(data)
        ) {
            throw new TypeError(
                '[NavigationGraph] Данные узла должны быть объектом.'
            )
        }

        if (this.nodes.has(id)) {
            throw new Error(
                `[NavigationGraph] Узел "${id}" уже существует.`
            )
        }

        const metadata = data.metadata ?? {}

        if (
            !metadata ||
            typeof metadata !== 'object' ||
            Array.isArray(metadata)
        ) {

        }

            zoneId: data.zoneId ?? null,

        this.nodes.set(id, node)
        this.edges.set(id, new Map())

        this.eventBus?.emit('navigation:node-added', {
            node: this.createNodeSnapshot(node)

    // --------------------------------------------------------

    connect(firstId, secondId, options = {}) {

        this.ensureActive()

        this.validateId(firstId)
        this.validateId(secondId)

        if (firstId === secondId) {
            throw new Error(
                '[NavigationGraph] Нельзя соединить узел с самим собой.'
            )
        }

        if (
            !options ||
            typeof options !== 'object' ||
            Array.isArray(options)
        ) {
            throw new TypeError(
                '[NavigationGraph] Параметры связи должны быть объектом.'
            )
        }

        if (
            !this.nodes.has(firstId) ||
            !this.nodes.has(secondId)
        ) {
            throw new Error(
                '[NavigationGraph] Оба узла должны существовать.'
            )
        }

        const cost = options.cost ?? 1
        const bidirectional = options.bidirectional ?? true

        if (!Number.isFinite(cost) || cost < 0) {
            throw new TypeError(
                '[NavigationGraph] Стоимость перехода должна быть неотрицательным числом.'
            )
        }

        if (typeof bidirectional !== 'boolean') {
            throw new TypeError(
                '[NavigationGraph] bidirectional должен быть логическим значением.'
            )
        }

        const firstConnections = this.edges.get(firstId)
        const secondConnections = this.edges.get(secondId)

        const existingForward = firstConnections.get(secondId)
        const existingReverse = secondConnections.get(firstId)

        const forwardIsCorrect =
            existingForward?.cost === cost

        const reverseIsCorrect =
            bidirectional
                ? existingReverse?.cost === cost
                : !existingReverse

        if (forwardIsCorrect && reverseIsCorrect) {
            return false
        }

        firstConnections.set(secondId, {
            from: firstId,
            to: secondId,
            cost
        })

        if (bidirectional) {

            secondConnections.set(firstId, {
                from: secondId,
                to: firstId,
                cost
            })

        } else {

            // Односторонняя связь не должна оставлять
            // старое автоматически созданное обратное ребро.

            secondConnections.delete(firstId)
        }

        this.eventBus?.emit('navigation:connected', {
            firstId,
            secondId,
            cost,
            bidirectional
        })

        return true
    }

    // --------------------------------------------------------
    // УДАЛЕНИЕ СВЯЗИ
    // --------------------------------------------------------

    disconnect(firstId, secondId, bidirectional = true) {

        this.ensureActive()

        this.validateId(firstId)
        this.validateId(secondId)

        if (typeof bidirectional !== 'boolean') {
            throw new TypeError(
                '[NavigationGraph] bidirectional должен быть логическим значением.'
            )
        }

        if (
            !this.nodes.has(firstId) ||
            !this.nodes.has(secondId)
        ) {
            return false
        }

        const firstRemoved =
            this.edges.get(firstId).delete(secondId)

        let secondRemoved = false

        if (bidirectional) {

            secondRemoved =
                this.edges.get(secondId).delete(firstId)
        }

        if (firstRemoved || secondRemoved) {

            this.eventBus?.emit('navigation:disconnected', {
                firstId,
                secondId,
                bidirectional
            })

            return true
        }

        return false
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ СОСЕДНИХ УЗЛОВ
    // --------------------------------------------------------

    getNeighbors(id) {

        this.ensureActive()

        const connections = this.edges.get(id)

        if (!connections) return []

        return Array.from(
            connections.values(),
            connection => ({
                node: this.getNode(connection.to),
                cost: connection.cost
            })
        )
    }

    // --------------------------------------------------------
    // ПРОВЕРКА СВЯЗИ
    // --------------------------------------------------------

    hasConnection(firstId, secondId) {

        this.ensureActive()

    }

    // ПОЛУЧЕНИЕ СТОИМОСТИ ПЕРЕХОДА

    getConnectionCost(firstId, secondId) {

    }

    // ОБНОВЛЕНИЕ СТОИМОСТИ ПЕРЕХОДА
    // --------------------------------------------------------

    setConnectionCost(firstId, secondId, cost, bidirectional = true) {

        this.ensureActive()

        this.validateId(firstId)

        if (typeof bidirectional !== 'boolean') {

        }

            this.edges.get(secondId)?.get(firstId)

                bidirectional &&

            )

        if (!changed) {

        }

        forward.cost = cost

        if (bidirectional && reverse) {

        }

        this.eventBus?.emit('navigation:cost-changed', {

    }

// ============================================================
// SAYLO GAME ENGINE
// NAVIGATION: PATH FINDER
// Алгоритм Дейкстры
// Версия архитектуры: 1.2
// ============================================================

export class PathFinder {

    constructor({
        graph,
        eventBus = null,
        maxVisitedNodes = 10000
    } = {}) {

        if (
            !graph ||
            typeof graph.getNode !== 'function' ||
            typeof graph.getNeighbors !== 'function'
        ) {
            throw new TypeError(
                '[PathFinder] Необходимо передать NavigationGraph.'
            )
        }

        this.validateLimit(maxVisitedNodes)

        this.graph = graph
        this.eventBus = eventBus
        this.maxVisitedNodes = maxVisitedNodes

        this.destroyed = false
    }

    // --------------------------------------------------------
    // ПОИСК МАРШРУТА
    // --------------------------------------------------------

    findPath(startId, goalId, options = {}) {

        this.ensureActive()
        this.validateId(startId)
        this.validateId(goalId)

        if (
            !options ||
            typeof options !== 'object' ||
            Array.isArray(options)
        ) {
            throw new TypeError(
                '[PathFinder] Параметры поиска должны быть объектом.'
            )
        }

        const blockedNodes = this.createBlockedSet(
            options.blockedNodes,
            'blockedNodes'
        )

        const blockedEdges = this.createBlockedSet(
            options.blockedEdges,
            'blockedEdges'
        )

        const maxVisitedNodes =
            options.maxVisitedNodes ?? this.maxVisitedNodes

        this.validateLimit(maxVisitedNodes)

        const startNode = this.graph.getNode(startId)
        const goalNode = this.graph.getNode(goalId)

        if (!startNode || !goalNode) {
            return this.createResult(
                false, [], Infinity, 0, 'node-not-found'
            )
        }

        if (
            blockedNodes.has(startId) ||
            blockedNodes.has(goalId)
        ) {
            return this.createResult(
                false, [], Infinity, 0, 'blocked-node'
            )
        }

        if (startId === goalId) {
            return this.createResult(
                true, [startId], 0, 0, null
            )
        }

        const openSet = new Set([startId])
        const closedSet = new Set()

        const cameFrom = new Map()
        const distance = new Map([[startId, 0]])

        let visitedNodes = 0

        while (openSet.size > 0) {

            if (visitedNodes >= maxVisitedNodes) {
                return this.createResult(
                    false, [], Infinity,
                    visitedNodes, 'visit-limit'
                )
            }

            const currentId = this.getLowestScoreNode(
                openSet,
                distance
            )

            if (currentId === null) break

            if (currentId === goalId) {
                return this.createResult(
                    true,
                    this.reconstructPath(cameFrom, currentId),
                    distance.get(currentId),
                    visitedNodes,
                    null
                )
            }

            openSet.delete(currentId)
            closedSet.add(currentId)
            visitedNodes++

            const neighbors = this.graph.getNeighbors(currentId)

            if (!Array.isArray(neighbors)) {
                throw new TypeError(
                    '[PathFinder] Некорректный список соседей.'
                )
            }

            for (const connection of neighbors) {

                if (
                    !connection ||
                    !connection.node ||
                    typeof connection.node.id !== 'string' ||
                    !Number.isFinite(connection.cost) ||
                    connection.cost < 0
                ) {
                    throw new TypeError(
                        '[PathFinder] Некорректное ребро графа.'
                    )
                }

                const neighborId = connection.node.id

                if (
                    blockedNodes.has(neighborId) ||
                    closedSet.has(neighborId)
                ) {
                    continue
                }

                if (
                    blockedEdges.has(
                        this.createEdgeKey(currentId, neighborId)
                    )
                ) {
                    continue
                }

                const newDistance =
                    distance.get(currentId) + connection.cost

                const oldDistance =
                    distance.get(neighborId) ?? Infinity

                if (newDistance >= oldDistance) continue

                cameFrom.set(neighborId, currentId)
                distance.set(neighborId, newDistance)

                openSet.add(neighborId)
            }
        }

        return this.createResult(
            false, [], Infinity, visitedNodes, 'no-path'
        )
    }

    // --------------------------------------------------------
    // ЗАБЛОКИРОВАННЫЕ УЗЛЫ И РЁБРА
    // --------------------------------------------------------

    createBlockedSet(value, name) {

        if (value === undefined || value === null) {
            return new Set()
        }

        if (
            typeof value === 'string' ||
            typeof value[Symbol.iterator] !== 'function'
        ) {
            throw new TypeError(
                `[PathFinder] ${name} должен быть итерируемым набором.`
            )
        }

        const result = new Set(value)

        for (const item of result) {
            if (
                typeof item !== 'string' ||
                item.trim().length === 0
            ) {
                throw new TypeError(
                    `[PathFinder] Некорректный элемент ${name}.`
                )
            }
        }

        return result
    }

    // --------------------------------------------------------
    // УЗЕЛ С МИНИМАЛЬНОЙ СТОИМОСТЬЮ
    // --------------------------------------------------------

    getLowestScoreNode(openSet, scores) {

        let bestNode = null
        let bestScore = Infinity

        for (const nodeId of openSet) {

            const score = scores.get(nodeId) ?? Infinity

            if (score < bestScore) {
                bestScore = score
                bestNode = nodeId
            }
        }

        return bestNode
    }

    // --------------------------------------------------------
    // ВОССТАНОВЛЕНИЕ МАРШРУТА
    // --------------------------------------------------------

    reconstructPath(cameFrom, currentId) {

        const path = [currentId]

        while (cameFrom.has(currentId)) {
            currentId = cameFrom.get(currentId)
            path.push(currentId)
        }

        path.reverse()

        return path
    }

    // --------------------------------------------------------
    // КЛЮЧ НАПРАВЛЕННОГО РЕБРА
    // --------------------------------------------------------

    createEdgeKey(firstId, secondId) {

        return JSON.stringify([firstId, secondId])
    }

    // --------------------------------------------------------
    // РЕЗУЛЬТАТ ПОИСКА
    // --------------------------------------------------------

    createResult(
        found,
        path,
        totalCost,
        visitedNodes,
        reason
    ) {

        const result = Object.freeze({
            found,
            path: Object.freeze([...path]),
            totalCost,
            visitedNodes,
            reason
        })

        this.eventBus?.emit(
            'navigation:path-search-completed',
            result
        )

        if (found) {
            this.eventBus?.emit(
                'navigation:path-found',
                result
            )
        }

        return result
    }

    // --------------------------------------------------------
    // ПРОВЕРКИ
    // --------------------------------------------------------

    validateId(id) {

        if (
            typeof id !== 'string' ||
            id.trim().length === 0
        ) {
            throw new TypeError(
                '[PathFinder] Некорректный идентификатор.'
            )
        }
    }

    validateLimit(value) {

        if (!Number.isInteger(value) || value < 1) {
            throw new TypeError(
                '[PathFinder] Лимит посещённых узлов должен быть положительным целым числом.'
            )
        }
    }

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[PathFinder] Модуль поиска маршрута уничтожен.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.graph = null
        this.eventBus = null
        this.destroyed = true
    }
}

        if (
            typeof id !== 'string' ||
            id.trim().length === 0

                '[Waypoints] Некорректный идентификатор.'

        this.eventBus = null
        this.destroyed = true
    }

        this.eventBus = null

    }

 // ============================================================
 // SAYLO GAME ENGINE
 // SYSTEM: ANIMATION
 // Централизованная система игровых анимаций
 // Совместимость: PixiJS 8 / Saylo Entity
 // Версия архитектуры: 2.0
 // ============================================================

export class AnimationSystem {

    constructor({
        eventBus = null,
        defaultEasing = 'easeInOut'
    } = {}) {

        if (
            eventBus !== null &&
            (
                typeof eventBus.on !== 'function' ||
                typeof eventBus.emit !== 'function'
            )
        ) {
            throw new TypeError('AnimationSystem: некорректный EventBus.')
        }

        this.eventBus = eventBus

        this.animations = new Map()
        this.targetAnimations = new WeakMap()

        this.nextAnimationId = 1

        this.defaultEasing = defaultEasing

        this.enabled = true
        this.destroyed = false

        this.easing = Object.freeze({
            linear: t => t,
            easeIn: t => t * t,
            easeOut: t => 1 - (1 - t) * (1 - t),
            easeInOut: t => (
                t < 0.5
                    ? 2 * t * t
                    : 1 - Math.pow(-2 * t + 2, 2) / 2
            ),
            easeInCubic: t => t * t * t,
            easeOutCubic: t => 1 - Math.pow(1 - t, 3),
            easeInOutCubic: t => (
                t < 0.5
                    ? 4 * t * t * t
                    : 1 - Math.pow(-2 * t + 2, 3) / 2
            ),
            easeOutBack: t => {
                const c1 = 1.70158
                const c3 = c1 + 1

                return 1 + c3 * Math.pow(t - 1, 3) +
                    c1 * Math.pow(t - 1, 2)
            },
            easeOutElastic: t => {
                if (t === 0 || t === 1) return t

                const c4 = (2 * Math.PI) / 3

                return Math.pow(2, -10 * t) *
                    Math.sin((t * 10 - 0.75) * c4) + 1
            }
        })

        if (!this.easing[this.defaultEasing]) {
            throw new Error(
                `AnimationSystem: неизвестная кривая "${defaultEasing}".`
            )
        }
    }

    // --------------------------------------------------------
    // СОЗДАНИЕ АНИМАЦИИ
    // --------------------------------------------------------

    animate(target, properties, {
        duration = 0.5,
        delay = 0,
        easing = this.defaultEasing,
        repeat = 0,
        yoyo = false,
        onUpdate = null,
        onComplete = null
    } = {}) {

        this.ensureActive()

        if (!this.enabled) {
            throw new Error('AnimationSystem: система отключена.')
        }

        if (!target || typeof target !== 'object') {
            throw new TypeError('AnimationSystem: некорректная цель.')
        }

        if (
            target.destroyed === true ||
            target.destroyed === 1
        ) {
            throw new Error('AnimationSystem: цель уже уничтожена.')
        }

        if (
            !properties ||
            typeof properties !== 'object' ||
            Array.isArray(properties)
        ) {
            throw new TypeError('AnimationSystem: properties должен быть объектом.')
        }

        if (
            !Number.isFinite(duration) ||
            duration < 0 ||
            !Number.isFinite(delay) ||
            delay < 0
        ) {
            throw new RangeError('AnimationSystem: некорректная длительность или задержка.')
        }

        if (
            repeat !== Infinity &&
            (!Number.isInteger(repeat) || repeat < 0)
        ) {
            throw new RangeError('AnimationSystem: repeat должен быть целым числом или Infinity.')
        }

        if (typeof yoyo !== 'boolean') {
            throw new TypeError('AnimationSystem: yoyo должен быть boolean.')
        }

        if (
            typeof onUpdate !== 'function' && onUpdate !== null
        ) {
            throw new TypeError('AnimationSystem: onUpdate должен быть функцией.')
        }

        if (
            typeof onComplete !== 'function' && onComplete !== null
        ) {
            throw new TypeError('AnimationSystem: onComplete должен быть функцией.')
        }

        const easingFunction = typeof easing === 'function'
            ? easing
            : this.easing[easing]

        if (typeof easingFunction !== 'function') {
            throw new Error(`AnimationSystem: неизвестная кривая "${easing}".`)
        }

        const tracks = this.createTracks(target, properties)

        if (tracks.length === 0) {
            throw new Error('AnimationSystem: не заданы свойства для анимации.')
        }

        const id = `animation-${this.nextAnimationId++}`

        const animation = {
            id,
            target,
            tracks,

            duration,
            delay,
            elapsed: 0,

            easing: easingFunction,

            repeat,
            completedRepeats: 0,
            yoyo,
            reversed: false,

            onUpdate,
            onComplete,

            paused: false,
            completed: false,
            cancelled: false
        }

        this.animations.set(id, animation)
        this.registerTargetAnimation(target, id)

        this.emit('animation:started', {
            animationId: id,
            targetId: this.getTargetId(target),
            duration,
            delay
        })

        if (duration === 0 && delay === 0) {
            this.applyProgress(animation, 1)
            this.completeAnimation(animation)
        }

        return id
    }

    // --------------------------------------------------------
    // СОЗДАНИЕ КАНАЛОВ АНИМАЦИИ
    // --------------------------------------------------------

    createTracks(target, properties) {
        const tracks = []

        for (const [property, endValue] of Object.entries(properties)) {
            if (!Number.isFinite(endValue)) {
                throw new TypeError(
                    `AnimationSystem: свойство "${property}" должно иметь числовое значение.`
                )
            }

            const startValue = this.readProperty(target, property)

            if (!Number.isFinite(startValue)) {
                throw new Error(
                    `AnimationSystem: невозможно прочитать свойство "${property}".`
                )
            }

            tracks.push({
                property,
                startValue,
                endValue
            })
        }

        return tracks
    }

    // --------------------------------------------------------
    // ЧТЕНИЕ СВОЙСТВ
    // --------------------------------------------------------

    readProperty(target, property) {

        if (
            (property === 'x' || property === 'y') &&
            target.position &&
            typeof target.setPosition === 'function'
        ) {
            return target.position[property]
        }

        if (property === 'scaleX') {
            return target.scale?.x
        }

        if (property === 'scaleY') {
            return target.scale?.y
        }

        return target[property]
    }

    // --------------------------------------------------------
    // ЗАПИСЬ СВОЙСТВ
    // --------------------------------------------------------

    writeProperty(target, property, value) {

        if (
            (property === 'x' || property === 'y') &&
            target.position &&
            typeof target.setPosition === 'function'
        ) {
            const position = target.position

            const x = property === 'x' ? value : position.x
            const y = property === 'y' ? value : position.y

            target.setPosition(x, y)

            return
        }

        if (property === 'scaleX') {
            if (!target.scale) {
                throw new Error('AnimationSystem: у цели отсутствует scale.')
            }

            target.scale.x = value
            return
        }

        if (property === 'scaleY') {
            if (!target.scale) {
                throw new Error('AnimationSystem: у цели отсутствует scale.')
            }

            target.scale.y = value
            return
        }

        target[property] = value
    }

    // --------------------------------------------------------
    // ОБНОВЛЕНИЕ ИГРОВОГО ЦИКЛА
    // deltaTime передаётся в секундах
    // --------------------------------------------------------

    update(deltaTime) {
        this.ensureActive()

        if (!this.enabled) return

        if (!Number.isFinite(deltaTime) || deltaTime < 0) {
            throw new RangeError('AnimationSystem: некорректный deltaTime.')
        }

        if (deltaTime === 0) return

        for (const animation of [...this.animations.values()]) {

            if (
                animation.paused ||
                animation.completed ||
                animation.cancelled
            ) {
                continue
            }

            if (animation.target.destroyed === true) {
                this.cancel(animation.id, 'target-destroyed')
                continue
            }

            if (animation.delay > 0) {
                const remainingDelay = animation.delay - animation.elapsed

                animation.elapsed += deltaTime

                if (remainingDelay > deltaTime) {
                    continue
                }

                const activeDelta = deltaTime - remainingDelay

                animation.elapsed = 0
                this.updateAnimation(animation, activeDelta)

                continue
            }

            this.updateAnimation(animation, deltaTime)
        }
    }

    updateAnimation(animation, deltaTime) {
        if (!this.animations.has(animation.id)) return

        let remainingTime = deltaTime

        // Защита от чрезмерного числа повторов за один кадр.
        let iterations = 0

        while (
            remainingTime >= 0 &&
            this.animations.has(animation.id) &&
            iterations < 100
        ) {
            iterations++

            const availableTime = animation.duration - animation.elapsed
            const step = Math.min(remainingTime, availableTime)

            animation.elapsed += step
            remainingTime -= step

            const progress = animation.duration === 0
                ? 1
                : Math.min(1, animation.elapsed / animation.duration)

            this.applyProgress(animation, progress)

            if (animation.elapsed < animation.duration) {
                break
            }

            if (animation.completedRepeats < animation.repeat) {
                animation.completedRepeats++

                animation.elapsed = 0

                if (animation.yoyo) {
                    animation.reversed = !animation.reversed
                } else {
                    this.applyProgress(animation, 0)
                }

                this.emit('animation:repeat', {
                    animationId: animation.id,
                    repeat: animation.completedRepeats
                })

                if (remainingTime <= 0) break

                continue
            }

            this.completeAnimation(animation)
            break
        }
    }

    // --------------------------------------------------------
    // ПРИМЕНЕНИЕ ПРОГРЕССА
    // --------------------------------------------------------

    applyProgress(animation, progress) {
        const easedProgress = animation.easing(
            Math.max(0, Math.min(1, progress))
        )

        for (const track of animation.tracks) {

            const start = animation.reversed
                ? track.endValue
                : track.startValue

            const end = animation.reversed
                ? track.startValue
                : track.endValue

            const value = start + (end - start) * easedProgress

            this.writeProperty(
                animation.target,
                track.property,
                value
            )
        }

        if (animation.onUpdate) {
            try {
                animation.onUpdate(
                    progress,
                    animation.target,
                    animation.id
                )
            } catch (error) {
                console.error('[Saylo Animation] Ошибка onUpdate:', error)
            }
        }

        this.emit('animation:updated', {
            animationId: animation.id,
            progress
        })
    }

    // --------------------------------------------------------
    // ЗАВЕРШЕНИЕ
    // --------------------------------------------------------

    completeAnimation(animation) {
        if (!this.animations.has(animation.id)) return

        animation.completed = true

        this.animations.delete(animation.id)
        this.unregisterTargetAnimation(animation.target, animation.id)

        if (animation.onComplete) {
            try {
                animation.onComplete(
                    animation.target,
                    animation.id
                )
            } catch (error) {
                console.error('[Saylo Animation] Ошибка onComplete:', error)
            }
        }

        this.emit('animation:completed', {
            animationId: animation.id,
            targetId: this.getTargetId(animation.target)
        })

        animation.target = null
        animation.tracks.length = 0
    }

    // --------------------------------------------------------
    // ОТМЕНА
    // --------------------------------------------------------

    cancel(animationId, reason = 'cancelled') {
        this.ensureActive()

        const animation = this.animations.get(animationId)

        if (!animation) return false

        animation.cancelled = true

        this.animations.delete(animationId)
        this.unregisterTargetAnimation(animation.target, animationId)

        this.emit('animation:cancelled', {
            animationId,
            targetId: this.getTargetId(animation.target),
            reason
        })

        animation.target = null
        animation.tracks.length = 0

        return true
    }

    cancelAll(reason = 'cancelled') {
        this.ensureActive()

        const ids = [...this.animations.keys()]

        for (const id of ids) {
            this.cancel(id, reason)
        }

        return ids.length
    }

    // --------------------------------------------------------
    // ПРИНУДИТЕЛЬНОЕ ЗАВЕРШЕНИЕ
    // --------------------------------------------------------

    finish(animationId) {
        this.ensureActive()

        const animation = this.animations.get(animationId)

        if (!animation) return false

        const finalValue = animation.yoyo && animation.reversed
            ? 0
            : 1

        this.applyProgress(animation, finalValue)
        this.completeAnimation(animation)

        return true
    }

    // --------------------------------------------------------
    // ПАУЗА И ПРОДОЛЖЕНИЕ
    // --------------------------------------------------------

    pause(animationId) {
        this.ensureActive()

        const animation = this.animations.get(animationId)

        if (!animation || animation.paused) return false

        animation.paused = true

        this.emit('animation:paused', {
            animationId
        })

        return true
    }

    resume(animationId) {
        this.ensureActive()

        const animation = this.animations.get(animationId)

        if (!animation || !animation.paused) return false

        animation.paused = false

        this.emit('animation:resumed', {
            animationId
        })

        return true
    }

    pauseAll() {
        this.ensureActive()

        for (const animation of this.animations.values()) {
            animation.paused = true
        }

        this.emit('animation:all-paused', {})
    }

    resumeAll() {
        this.ensureActive()

        for (const animation of this.animations.values()) {
            animation.paused = false
        }

        this.emit('animation:all-resumed', {})
    }

    // --------------------------------------------------------
    // АНИМАЦИИ КОНКРЕТНОЙ ЦЕЛИ
    // --------------------------------------------------------

    cancelTarget(target, reason = 'target-cancelled') {
        this.ensureActive()

        const ids = this.targetAnimations.get(target)

        if (!ids) return 0

        const list = [...ids]

        for (const id of list) {
            this.cancel(id, reason)
        }

        return list.length
    }

    registerTargetAnimation(target, animationId) {
        let ids = this.targetAnimations.get(target)

        if (!ids) {
            ids = new Set()
            this.targetAnimations.set(target, ids)
        }

        ids.add(animationId)
    }

    unregisterTargetAnimation(target, animationId) {
        const ids = this.targetAnimations.get(target)

        if (!ids) return

        ids.delete(animationId)

        if (ids.size === 0) {
            this.targetAnimations.delete(target)
        }
    }

    // --------------------------------------------------------
    // СОСТОЯНИЕ
    // --------------------------------------------------------

    getAnimation(animationId) {
        this.ensureActive()

        const animation = this.animations.get(animationId)

        if (!animation) return null

        return Object.freeze({
            id: animation.id,
            duration: animation.duration,
            delay: animation.delay,
            elapsed: animation.elapsed,
            repeat: animation.repeat,
            completedRepeats: animation.completedRepeats,
            paused: animation.paused,
            reversed: animation.reversed,
            targetId: this.getTargetId(animation.target)
        })
    }

    getActiveCount() {
        this.ensureActive()
        return this.animations.size
    }

    getAnimations() {
        this.ensureActive()

// ============================================================
// SAYLO GAME ENGINE
// SYSTEM: INPUT
// Управление клавиатурой, мышью и сенсорным вводом
// Версия архитектуры: 1.0
// ============================================================

export class InputSystem {

    constructor({
        target = window,
        pointerTarget = null,
        eventBus = null,
        preventDefaultKeys = []
    } = {}) {

        if (!target || typeof target.addEventListener !== 'function') {
            throw new TypeError('InputSystem: target должен поддерживать addEventListener.')
        }

        if (
            pointerTarget !== null &&
            typeof pointerTarget.addEventListener !== 'function'
        ) {
            throw new TypeError('InputSystem: некорректный pointerTarget.')
        }

        if (
            !Array.isArray(preventDefaultKeys) ||
            preventDefaultKeys.some(key => typeof key !== 'string')
        ) {
            throw new TypeError('InputSystem: preventDefaultKeys должен быть массивом строк.')
        }

        this.target = target
        this.pointerTarget = pointerTarget
        this.eventBus = eventBus

        this.preventDefaultKeys = new Set(preventDefaultKeys)

        this.pressedKeys = new Set()
        this.pointers = new Map()

        this.enabled = true
        this.attached = false
        this.destroyed = false

        this.handleKeyDown = this.onKeyDown.bind(this)
        this.handleKeyUp = this.onKeyUp.bind(this)
        this.handleBlur = this.onBlur.bind(this)

        this.handlePointerDown = this.onPointerDown.bind(this)
        this.handlePointerMove = this.onPointerMove.bind(this)
        this.handlePointerUp = this.onPointerUp.bind(this)
        this.handlePointerCancel = this.onPointerCancel.bind(this)
    }

    attach() {
        this.ensureActive()

        if (this.attached) return false

        this.target.addEventListener('keydown', this.handleKeyDown)
        this.target.addEventListener('keyup', this.handleKeyUp)
        this.target.addEventListener('blur', this.handleBlur)

        if (this.pointerTarget) {
            this.pointerTarget.addEventListener('pointerdown', this.handlePointerDown)
            this.pointerTarget.addEventListener('pointermove', this.handlePointerMove)
            this.pointerTarget.addEventListener('pointerup', this.handlePointerUp)
            this.pointerTarget.addEventListener('pointercancel', this.handlePointerCancel)
        }

        this.attached = true

        this.emit('input:attached', {})

        return true
    }

    detach() {
        if (!this.attached) return false

        this.target.removeEventListener('keydown', this.handleKeyDown)
        this.target.removeEventListener('keyup', this.handleKeyUp)
        this.target.removeEventListener('blur', this.handleBlur)

        if (this.pointerTarget) {
            this.pointerTarget.removeEventListener('pointerdown', this.handlePointerDown)
            this.pointerTarget.removeEventListener('pointermove', this.handlePointerMove)
            this.pointerTarget.removeEventListener('pointerup', this.handlePointerUp)
            this.pointerTarget.removeEventListener('pointercancel', this.handlePointerCancel)
        }

        this.pressedKeys.clear()
        this.pointers.clear()

        this.attached = false

        this.emit('input:detached', {})

        return true
    }

    onKeyDown(event) {
        if (!this.enabled || this.destroyed) return

        const key = event.code || event.key

        if (this.preventDefaultKeys.has(key)) {
            event.preventDefault()
        }

        const alreadyPressed = this.pressedKeys.has(key)

        this.pressedKeys.add(key)

        if (!alreadyPressed) {
            this.emit('input:key-down', {
                key,
                code: event.code || null,
                repeat: false
            })
        }
    }

    onKeyUp(event) {
        if (!this.enabled || this.destroyed) return

        const key = event.code || event.key

        this.pressedKeys.delete(key)

        this.emit('input:key-up', {
            key,
            code: event.code || null
        })
    }

    onBlur() {
        if (this.destroyed) return

        const releasedKeys = [...this.pressedKeys]

        this.pressedKeys.clear()
        this.pointers.clear()

        for (const key of releasedKeys) {
            this.emit('input:key-up', {
                key,
                code: null,
                reason: 'window-blur'
            })
        }

        this.emit('input:reset', {
            reason: 'window-blur'
        })
    }

    onPointerDown(event) {
        this.handlePointerEvent('input:pointer-down', event, true)
    }

    onPointerMove(event) {
        this.handlePointerEvent('input:pointer-move', event, false)
    }

    onPointerUp(event) {
        this.handlePointerEvent('input:pointer-up', event, false)

        this.pointers.delete(event.pointerId)
    }

    onPointerCancel(event) {
        this.handlePointerEvent('input:pointer-cancel', event, false)

        this.pointers.delete(event.pointerId)
    }

    handlePointerEvent(eventName, event, pressed) {
        if (!this.enabled || this.destroyed) return

        const rect = this.pointerTarget?.getBoundingClientRect()

        if (!rect) return

        const pointer = {
            pointerId: event.pointerId,
            pointerType: event.pointerType,
            button: event.button,
            buttons: event.buttons,

            clientX: event.clientX,
            clientY: event.clientY,

            x: event.clientX - rect.left,
            y: event.clientY - rect.top,

            width: event.width,
            height: event.height,

            pressure: event.pressure,
            isPrimary: event.isPrimary
        }

        if (pressed) {
            this.pointers.set(event.pointerId, pointer)
        } else if (this.pointers.has(event.pointerId)) {
            this.pointers.set(event.pointerId, pointer)
        }

        this.emit(eventName, pointer)
    }

    isKeyDown(key) {
        this.ensureActive()

        if (typeof key !== 'string') {
            throw new TypeError('InputSystem: key должен быть строкой.')
        }

        return this.pressedKeys.has(key)
    }

    getPressedKeys() {
        this.ensureActive()

        return Object.freeze([...this.pressedKeys])
    }

    getPointer(pointerId) {
        this.ensureActive()

        const pointer = this.pointers.get(pointerId)

        return pointer ? Object.freeze({ ...pointer }) : null
    }

    setEnabled(enabled) {
        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError('InputSystem: enabled должен быть boolean.')
        }

        if (this.enabled === enabled) return false

        this.enabled = enabled

        if (!enabled) {
            this.onBlur()
        }

        this.emit('input:enabled-changed', { enabled })

        return true
    }

    emit(eventName, payload) {
        if (this.eventBus && typeof this.eventBus.emit === 'function') {
            this.eventBus.emit(eventName, payload)
        }
    }

    ensureActive() {
        if (this.destroyed) {
            throw new Error('InputSystem: система уже уничтожена.')
        }
    }

    destroy() {
        if (this.destroyed) return

        this.detach()

        this.pressedKeys.clear()
        this.pointers.clear()

        this.target = null
        this.pointerTarget = null
        this.eventBus = null

        this.destroyed = true
    }
}

    hitTest(record, point) {
        const entity = record.entity

        if (
            entity.destroyed ||
            entity.visible === false ||
            entity.active === false ||
            !record.interactive
        ) {
            return false
        }

        if (typeof record.hitArea === 'function') {
            return Boolean(record.hitArea(point, entity))
        }

        const position = entity.position

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y)
        ) {
            return false
        }

        const area = record.hitArea

        const width = area ? area.width : entity.width
        const height = area ? area.height : entity.height

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width < 0 ||
            height < 0
        ) {
            return false
        }

        const offsetX = area ? area.x : -width / 2
        const offsetY = area ? area.y : -height / 2

        const left = position.x + offsetX
        const top = position.y + offsetY

        return (
            point.x >= left &&
            point.x <= left + width &&
            point.y >= top &&
            point.y <= top + height
        )
    }

    getEntityAt(point) {
        this.ensureActive()

        if (
            !point ||
            !Number.isFinite(point.x) ||
            !Number.isFinite(point.y)
        ) {
            throw new TypeError('InteractionSystem: некорректная точка поиска.')
        }

        const records = [...this.entities.values()].sort((a, b) => {
            if (a.zIndex !== b.zIndex) {
                return b.zIndex - a.zIndex
            }

            return b.order - a.order
        })

        for (const record of records) {
            if (this.hitTest(record, point)) {
                return record.entity
            }
        }

        return null
    }

        const point = this.toWorldPosition(pointer)

        const state = {
            pointerId: pointer.pointerId,
            entityId: entity?.id || null,

            startX: pointer.x,
            startY: pointer.y,

            startWorldX: point.x,
            startWorldY: point.y,

            lastWorldX: point.x,
            lastWorldY: point.y,

            dragging: false

        if (!entity) {
            this.setSelectedEntity(null)

            this.emit('interaction:empty-press', {
                pointerId: pointer.pointerId,
                position: point
            })

            return
        }

        this.setSelectedEntity(entity.id)

        this.emit('interaction:pointer-down', {
            entityId: entity.id,
            entity,
            pointerId: pointer.pointerId,
            position: point,
            pointer
        })
    }

    // --------------------------------------------------------
    // ДВИЖЕНИЕ И ПЕРЕТАСКИВАНИЕ
    // --------------------------------------------------------

    onPointerMove(pointer) {
        if (!this.enabled || this.destroyed) return

        const point = this.toWorldPosition(pointer)
        const state = this.pointers.get(pointer.pointerId)

        const hovered = this.getEntityAt(point)

        this.setHoveredEntity(hovered?.id || null)

        if (!state || !state.entityId) return

        const record = this.entities.get(state.entityId)

        if (!record) {
            this.pointers.delete(pointer.pointerId)
            return
        }

        const distance = Math.hypot(
            pointer.x - state.startX,
            pointer.y - state.startY
        )

        if (
            !state.dragging &&
            record.draggable &&
            distance >= this.dragThreshold
        ) {
            state.dragging = true

            this.emit('interaction:drag-start', {
                entityId: record.entity.id,
                entity: record.entity,
                pointerId: pointer.pointerId,
                position: point
            })
        }

        if (state.dragging) {
            const position = record.entity.position

            const deltaX = point.x - state.lastWorldX
            const deltaY = point.y - state.lastWorldY

            record.entity.setPosition(
                position.x + deltaX,
                position.y + deltaY
            )

            this.emit('interaction:drag', {
                entityId: record.entity.id,
                entity: record.entity,
                pointerId: pointer.pointerId,
                position: point,
                delta: {
                    x: deltaX,
                    y: deltaY
                }
            })
        }

        state.lastWorldX = point.x
        state.lastWorldY = point.y
    }

    // --------------------------------------------------------
    // ОТПУСКАНИЕ
    // --------------------------------------------------------

    onPointerUp(pointer) {
        if (this.destroyed) return

        const state = this.pointers.get(pointer.pointerId)

        if (!state) return

        const point = this.toWorldPosition(pointer)

        const record = state.entityId
            ? this.entities.get(state.entityId)
            : null

        if (record) {
            if (state.dragging) {
                this.emit('interaction:drag-end', {
                    entityId: record.entity.id,
                    entity: record.entity,
                    pointerId: pointer.pointerId,
                    position: point
                })
            } else {
                const distance = Math.hypot(
                    pointer.x - state.startX,
                    pointer.y - state.startY
                )

                const releasedEntity = this.getEntityAt(point)

                const primaryButton =
                    pointer.pointerType === 'touch' ||
                    pointer.button === 0

                if (
                    primaryButton &&
                    releasedEntity?.id === state.entityId &&
                    distance <= this.clickThreshold
                ) {
                    this.emit('interaction:click', {
                        entityId: record.entity.id,
                        entity: record.entity,
                        pointerId: pointer.pointerId,
                        position: point
                    })
                }
            }

            this.emit('interaction:pointer-up', {
                entityId: record.entity.id,
                entity: record.entity,
                pointerId: pointer.pointerId,
                position: point
            })
        }

        this.pointers.delete(pointer.pointerId)
    }

    // --------------------------------------------------------
    // ОТМЕНА УКАЗАТЕЛЯ
    // --------------------------------------------------------

    onPointerCancel(pointer) {
        if (this.destroyed) return

        this.cancelPointer(pointer.pointerId, 'pointer-cancel')
    }

    cancelPointer(pointerId, reason = 'cancelled') {
        const state = this.pointers.get(pointerId)

        if (!state) return false

        const record = state.entityId
            ? this.entities.get(state.entityId)
            : null

        if (record && state.dragging) {
            this.emit('interaction:drag-cancel', {
                entityId: record.entity.id,
                entity: record.entity,
                pointerId,
                reason
            })
        }

        this.pointers.delete(pointerId)

        return true
    }

    cancelAllPointers(reason = 'cancelled') {
        for (const pointerId of [...this.pointers.keys()]) {
            this.cancelPointer(pointerId, reason)
        }
    }

    onInputReset() {
        this.cancelAllPointers('input-reset')
        this.setHoveredEntity(null)
    }

    onInputDetached() {
        this.cancelAllPointers('input-detached')
        this.setHoveredEntity(null)
    }

    // --------------------------------------------------------
    // НАВЕДЕНИЕ
    // --------------------------------------------------------

    setHoveredEntity(entityId) {
        if (this.hoveredEntityId === entityId) return false

        const previousId = this.hoveredEntityId
        const previous = previousId
            ? this.entities.get(previousId)
            : null

        if (previous) {
            this.emit('interaction:pointer-out', {
                entityId: previousId,
                entity: previous.entity
            })
        }

        this.hoveredEntityId = entityId

        const current = entityId
            ? this.entities.get(entityId)
            : null

        if (current) {
            this.emit('interaction:pointer-over', {
                entityId,
                entity: current.entity
            })
        }

        return true
    }

    getHoveredEntity() {
        this.ensureActive()

        if (!this.hoveredEntityId) return null

        return this.entities.get(this.hoveredEntityId)?.entity || null
    }

    // --------------------------------------------------------
    // ВЫБОР ОБЪЕКТА
    // --------------------------------------------------------

    setSelectedEntity(entityId) {
        this.ensureActive()

        if (
            entityId !== null &&
            !this.entities.has(entityId)
        ) {
            throw new Error(`InteractionSystem: Entity "${entityId}" не зарегистрирован.`)
        }

        if (this.selectedEntityId === entityId) return false

        const previousId = this.selectedEntityId

        this.selectedEntityId = entityId

        this.emit('interaction:selection-changed', {
            previousEntityId: previousId,
            entityId
        })

        return true
    }

    getSelectedEntity() {
        this.ensureActive()

        if (!this.selectedEntityId) return null

        return this.entities.get(this.selectedEntityId)?.entity || null
    }

    // --------------------------------------------------------
    // УПРАВЛЕНИЕ СИСТЕМОЙ
    // --------------------------------------------------------

    setEnabled(enabled) {
        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError('InteractionSystem: enabled должен быть boolean.')
        }

        if (this.enabled === enabled) return false

        this.enabled = enabled

        if (!enabled) {
            this.cancelAllPointers('interaction-disabled')
            this.setHoveredEntity(null)
        }

        this.emit('interaction:enabled-changed', {
            enabled
        })

        return true
    }

    emit(eventName, payload) {
        if (this.eventBus) {
            this.eventBus.emit(eventName, payload)
        }
    }

    ensureActive() {
        if (this.destroyed) {
            throw new Error('InteractionSystem: система уже уничтожена.')
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {
        if (this.destroyed) return

        this.detach()

        this.entities.clear()
        this.pointers.clear()

        this.hoveredEntityId = null
        this.selectedEntityId = null

        this.inputSystem = null
        this.camera = null
        this.eventBus = null

        this.destroyed = true
    }

const DEFAULT_CONFIG = Object.freeze({

    initialize() {

            return false
        }

        this.validateDeltaTime(deltaTime)

        if (deltaTime === 0) {
            return false
        }

        const targetFrameTime = 1000 / this.config.targetFPS

        this.frameCount++
        this.sampleFrames++

        this.totalElapsed += deltaTime
        this.sampleElapsed += deltaTime

        this.currentFrameTime = frameTime

        this.frameTimes.push(frameTime)

        if (this.frameTimes.length > this.config.historySize) {
            this.frameTimes.shift()
        }

        this.minFrameTime = Math.min(
            this.minFrameTime,
            frameTime
        )

        this.maxFrameTime = Math.max(
            this.maxFrameTime,
            frameTime
        )

        if (frameTime > targetFrameTime * this.config.slowFrameThreshold) {
            this.slowFrames++
        }

        if (frameTime > targetFrameTime * this.config.criticalFrameThreshold) {
            this.criticalFrames++
        }

        if (frameTime > targetFrameTime * 1.5) {
            this.droppedFrames++
        }

        this.averageFrameTime = this.calculateAverage(
            this.frameTimes
        )

        this.updateMemoryMetrics()
        this.updateAdaptiveQuality(deltaTime)

        if (this.sampleElapsed >= this.config.sampleInterval) {
            this.updateSample()
        }

        if (this.emitElapsed >= this.config.emitInterval) {
            this.emitElapsed = 0
            this.emitMetrics()
        }

        return true
    }

    // --------------------------------------------------------
    // FRAME SAMPLING
    // --------------------------------------------------------

    updateSample() {
        if (this.sampleElapsed <= 0) {
            return
        }

        this.currentFPS = this.sampleFrames / this.sampleElapsed

        this.averageFPS = this.totalElapsed > 0
            ? this.frameCount / this.totalElapsed
            : 0

        const sample = Object.freeze({
            timestamp: Date.now(),
            fps: this.currentFPS,
            frameTime: this.averageFrameTime,
            frames: this.sampleFrames
        })

        this.frameHistory.push(sample)

        if (this.frameHistory.length > this.config.historySize) {
            this.frameHistory.shift()
        }

        this.emit('performance:sample', sample)

        this.sampleElapsed = 0
        this.sampleFrames = 0
    }

    // --------------------------------------------------------
    // SYSTEM TIMING
    // Измерение времени отдельных игровых систем.
    // Время передаётся в миллисекундах.
    // --------------------------------------------------------

    recordSystemTime(name, duration) {
        this.ensureActive()

        if (!this.enabled || this.paused) {
            return false
        }

        if (
            typeof name !== 'string' ||
            !name.trim()
        ) {
            throw new TypeError(
                '[PerformanceSystem] Имя системы должно быть строкой.'
            )
        }

        if (
            typeof duration !== 'number' ||
            !Number.isFinite(duration) ||
            duration < 0
        ) {
            throw new RangeError(
                '[PerformanceSystem] Некорректное время выполнения.'
            )
        }

        const key = name.trim()

        let metric = this.systemMetrics.get(key)

        if (!metric) {
            metric = {
                name: key,
                count: 0,
                total: 0,
                min: Infinity,
                max: 0,
                average: 0,
                last: 0
            }

            this.systemMetrics.set(key, metric)
        }

        metric.count++
        metric.total += duration
        metric.last = duration

        metric.min = Math.min(
            metric.min,
            duration
        )

        metric.max = Math.max(
            metric.max,
            duration
        )

        metric.average = metric.total / metric.count

        return true
    }

    getSystemMetrics() {
        this.ensureActive()

        return Array.from(
            this.systemMetrics.values(),
            metric => Object.freeze({
                ...metric
            })
        )
    }

    getSystemMetric(name) {
        this.ensureActive()

        if (typeof name !== 'string') {
            throw new TypeError(
                '[PerformanceSystem] Имя системы должно быть строкой.'
            )
        }

        const metric = this.systemMetrics.get(name)

        return metric
            ? Object.freeze({ ...metric })
            : null
    }

    resetSystemMetrics() {
        this.ensureActive()

        this.systemMetrics.clear()

        this.emit('performance:systems-reset')

        return true
    }

    // --------------------------------------------------------
    // MEMORY MONITORING
    // --------------------------------------------------------

    updateMemoryMetrics() {
        if (!this.config.memoryMonitoring) {
            this.lastMemory = null
            return
        }

        try {
            const memory = globalThis.performance?.memory

            if (!memory) {
                this.lastMemory = null
                return
            }

            this.lastMemory = Object.freeze({
                usedJSHeapSize: memory.usedJSHeapSize,
                totalJSHeapSize: memory.totalJSHeapSize,
                jsHeapSizeLimit: memory.jsHeapSizeLimit
            })

        } catch {
            this.lastMemory = null
        }
    }

    getMemoryMetrics() {
        this.ensureActive()

        return this.lastMemory
            ? Object.freeze({ ...this.lastMemory })
            : null
    }

    // --------------------------------------------------------
    // ADAPTIVE QUALITY
    // Система выдаёт рекомендацию движку,
    // но не меняет качество рендеринга самостоятельно.
    // --------------------------------------------------------

    updateAdaptiveQuality(deltaTime) {
        if (!this.config.adaptiveQuality) {
            return
        }

        const targetFrameTime = 1000 / this.config.targetFPS

        const recentAverage = this.calculateAverage(
            this.frameTimes.slice(-30)
        )

        const estimatedFPS = recentAverage > 0
            ? 1000 / recentAverage
            : this.config.targetFPS

        const critical =
            estimatedFPS < this.config.criticalFPSThreshold ||
            this.currentFrameTime > targetFrameTime * this.config.criticalFrameThreshold

        const low =
            estimatedFPS < this.config.lowFPSThreshold ||
            this.currentFrameTime > targetFrameTime * this.config.slowFrameThreshold

        if (critical) {
            this.lowFPSDuration += deltaTime
        } else if (low) {
            this.lowFPSDuration += deltaTime * 0.5
        } else {
            this.lowFPSDuration = Math.max(
                0,
                this.lowFPSDuration - deltaTime
            )
        }

        if (this.lowFPSDuration >= 2) {
            this.recoveryElapsed = 0

            this.setRecommendedQuality(
                'low',
                'sustained_low_performance'
            )

            return
        }

        if (this.lowFPSDuration >= 1) {
            this.recoveryElapsed = 0

            this.setRecommendedQuality(
                'medium',
                'reduced_frame_rate'
            )

            return
        }

        if (
            estimatedFPS >= this.config.recoveryFPSThreshold &&
            this.currentFrameTime <= targetFrameTime * 1.2
        ) {
            this.recoveryElapsed += deltaTime

            if (
                this.recoveryElapsed >= this.config.recoveryDuration
            ) {
                this.lowFPSDuration = 0

                this.setRecommendedQuality(
                    'high',
                    'performance_recovered'
                )
            }
        } else {
            this.recoveryElapsed = 0
        }
    }

    setRecommendedQuality(level, reason = 'manual') {
        this.ensureActive()

        const allowed = ['low', 'medium', 'high']

        if (!allowed.includes(level)) {
            throw new RangeError(
                '[PerformanceSystem] Неизвестный уровень качества.'
            )
        }

        if (this.recommendedQuality === level) {
            return false
        }

        const previous = this.recommendedQuality

        this.recommendedQuality = level

        this.emit('performance:quality-recommended', {
            previous,
            level,
            reason
        })

        return true
    }

    setQualityLevel(level) {
        this.ensureActive()

        const allowed = ['low', 'medium', 'high']

        if (!allowed.includes(level)) {
            throw new RangeError(
                '[PerformanceSystem] Неизвестный уровень качества.'
            )
        }

        if (this.qualityLevel === level) {
            return false
        }

        const previous = this.qualityLevel

        this.qualityLevel = level

        this.emit('performance:quality-changed', {
            previous,
            level
        })

        return true
    }

    getQualityState() {
        this.ensureActive()

        return Object.freeze({
            current: this.qualityLevel,
            recommended: this.recommendedQuality,
            adaptive: this.config.adaptiveQuality
        })
    }

    // --------------------------------------------------------
    // WARNINGS
    // --------------------------------------------------------

    getWarnings() {
        this.ensureActive()

        const warnings = []

        const targetFrameTime = 1000 / this.config.targetFPS

        if (
            this.currentFrameTime >
            targetFrameTime * this.config.criticalFrameThreshold
        ) {
            warnings.push({
                type: 'critical_frame',
                value: this.currentFrameTime
            })
        } else if (
            this.currentFrameTime >
            targetFrameTime * this.config.slowFrameThreshold
        ) {
            warnings.push({
                type: 'slow_frame',
                value: this.currentFrameTime
            })
        }

        if (
            this.currentFPS > 0 &&
            this.currentFPS < this.config.criticalFPSThreshold
        ) {
            warnings.push({
                type: 'critical_fps',
                value: this.currentFPS
            })
        } else if (
            this.currentFPS > 0 &&
            this.currentFPS < this.config.lowFPSThreshold
        ) {
            warnings.push({
                type: 'low_fps',
                value: this.currentFPS
            })
        }

        return warnings.map(warning =>
            Object.freeze({ ...warning })
        )
    }

    // --------------------------------------------------------
    // STATISTICS
    // --------------------------------------------------------

    getStats() {
        this.ensureActive()

        const sorted = [...this.frameTimes].sort(
            (a, b) => a - b
        )

        return Object.freeze({
            enabled: this.enabled,
            paused: this.paused,

            frames: this.frameCount,
            elapsed: this.totalElapsed,

            fps: this.currentFPS,
            averageFPS: this.averageFPS,

            frameTime: this.currentFrameTime,
            averageFrameTime: this.averageFrameTime,

            minFrameTime: this.frameTimes.length
                ? this.minFrameTime
                : 0,

            maxFrameTime: this.maxFrameTime,

            p95FrameTime: this.calculatePercentile(sorted, 0.95),

            droppedFrames: this.droppedFrames,
            slowFrames: this.slowFrames,
            criticalFrames: this.criticalFrames,

            memory: this.getMemoryMetrics(),

            quality: this.getQualityState(),

            systems: this.getSystemMetrics(),

            warnings: this.getWarnings()
        })
    }

    getHistory() {
        this.ensureActive()

        return this.frameHistory.map(sample =>
            Object.freeze({ ...sample })
        )
    }

    getConfig() {
        this.ensureActive()

        return Object.freeze({
            ...this.config
        })
    }

    // --------------------------------------------------------
    // CONFIGURATION
    // --------------------------------------------------------

    setTargetFPS(fps) {
        this.ensureActive()

        if (
            typeof fps !== 'number' ||
            !Number.isFinite(fps) ||
            fps < 1 ||
            fps > 240
        ) {
            throw new RangeError(
                '[PerformanceSystem] targetFPS должен быть от 1 до 240.'
            )
        }

        const previous = this.config.targetFPS

        this.config.targetFPS = fps

        this.emit('performance:target-fps-changed', {
            previous,
            value: fps
        })

        return true
    }

    setAdaptiveQuality(enabled) {
        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError(
                '[PerformanceSystem] enabled должен быть boolean.'
            )
        }

        this.config.adaptiveQuality = enabled

        if (!enabled) {
            this.lowFPSDuration = 0
            this.recoveryElapsed = 0
        }

        this.emit('performance:adaptive-quality-changed', {
            enabled
        })

        return true
    }

    // --------------------------------------------------------
    // ENABLE / PAUSE
    // --------------------------------------------------------

    setEnabled(enabled) {
        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError(
                '[PerformanceSystem] enabled должен быть boolean.'
            )
        }

        if (this.enabled === enabled) {
            return false
        }

        this.enabled = enabled

        this.emit('performance:enabled-changed', {
            enabled
        })

        return true
    }

    pause() {
        this.ensureActive()

        if (this.paused) {
            return false
        }

        this.paused = true

        this.emit('performance:paused')

        return true
    }

    resume() {
        this.ensureActive()

        if (!this.paused) {
            return false
        }

        this.paused = false

        this.emit('performance:resumed')

        return true
    }

    // --------------------------------------------------------
    // RESET
    // --------------------------------------------------------

    resetStats() {
        this.ensureActive()

        this.frameCount = 0
        this.totalElapsed = 0

        this.sampleElapsed = 0
        this.sampleFrames = 0
        this.emitElapsed = 0

        this.frameTimes = []
        this.frameHistory = []

        this.droppedFrames = 0
        this.slowFrames = 0
        this.criticalFrames = 0

        this.currentFPS = 0
        this.averageFPS = 0

        this.currentFrameTime = 0
        this.averageFrameTime = 0
        this.minFrameTime = Infinity
        this.maxFrameTime = 0

        this.lowFPSDuration = 0
        this.recoveryElapsed = 0

        this.lastWarning = null
        this.lastMemory = null

        this.systemMetrics.clear()

        this.emit('performance:stats-reset')

        return true
    }

    // --------------------------------------------------------
    // EVENTS
    // --------------------------------------------------------

    emitMetrics() {
        const stats = this.getStats()

        this.emit('performance:metrics', stats)

        const warnings = stats.warnings

        if (warnings.length > 0) {
            const signature = warnings
                .map(item => item.type)
                .join('|')

            if (signature !== this.lastWarning) {
                this.lastWarning = signature

                this.emit('performance:warning', {
                    warnings
                })
            }
        } else if (this.lastWarning !== null) {
            this.lastWarning = null

            this.emit('performance:recovered', {
                fps: this.currentFPS
            })
        }
    }

    // --------------------------------------------------------
    // VALIDATION / HELPERS
    // --------------------------------------------------------

    validateConfig(config) {
        if (
            !config ||
            typeof config !== 'object' ||
            Array.isArray(config)
        ) {
            throw new TypeError(
                '[PerformanceSystem] config должен быть объектом.'
            )
        }

        const merged = {
            ...DEFAULT_CONFIG,
            ...config
        }

        const booleanKeys = [
            'enabled',
            'adaptiveQuality',
            'memoryMonitoring'
        ]

        for (const key of booleanKeys) {
            if (typeof merged[key] !== 'boolean') {
                throw new TypeError(
                    `[PerformanceSystem] ${key} должен быть boolean.`
                )
            }
        }

        const positiveKeys = [
            'targetFPS',
            'sampleInterval',
            'historySize',
            'slowFrameThreshold',
            'criticalFrameThreshold',
            'lowFPSThreshold',
            'criticalFPSThreshold',
            'recoveryFPSThreshold',
            'recoveryDuration',
            'emitInterval'
        ]

        for (const key of positiveKeys) {
            const value = merged[key]

            if (
                typeof value !== 'number' ||
                !Number.isFinite(value) ||
                value <= 0
            ) {
                throw new RangeError(
                    `[PerformanceSystem] Некорректное значение ${key}.`
                )
            }
        }

        if (
            !Number.isInteger(merged.historySize) ||
            merged.historySize < 2
        ) {
            throw new RangeError(
                '[PerformanceSystem] historySize должен быть целым числом >= 2.'
            )
        }

        if (merged.targetFPS > 240) {
            throw new RangeError(
                '[PerformanceSystem] targetFPS не может превышать 240.'
            )
        }

        if (
            merged.criticalFrameThreshold <
            merged.slowFrameThreshold
        ) {
            throw new RangeError(
                '[PerformanceSystem] criticalFrameThreshold должен быть не меньше slowFrameThreshold.'
            )
        }

        if (
            merged.criticalFPSThreshold >
            merged.lowFPSThreshold
        ) {
            throw new RangeError(
                '[PerformanceSystem] criticalFPSThreshold не может превышать lowFPSThreshold.'
            )
        }

        if (
            merged.recoveryFPSThreshold <
            merged.lowFPSThreshold
        ) {
            throw new RangeError(
                '[PerformanceSystem] recoveryFPSThreshold должен быть не меньше lowFPSThreshold.'
            )
        }
    }

    validateDeltaTime(deltaTime) {
        if (
            typeof deltaTime !== 'number' ||
            !Number.isFinite(deltaTime) ||
            deltaTime < 0
        ) {
            throw new RangeError(
                '[PerformanceSystem] deltaTime должен быть неотрицательным числом в секундах.'
            )
        }
    }

    calculateAverage(values) {
        if (!values.length) {
            return 0
        }

        let total = 0

        for (const value of values) {
            total += value
        }

        return total / values.length
    }

    calculatePercentile(sortedValues, percentile) {
        if (!sortedValues.length) {
            return 0
        }

        const index = Math.min(
            sortedValues.length - 1,
            Math.ceil(sortedValues.length * percentile) - 1
        )

        return sortedValues[index]
    }

    ensureActive() {
        if (this.destroyed) {
            throw new Error(
                '[PerformanceSystem] Модуль уже уничтожен.'
            )
        }
    }

    emit(eventName, payload = {}) {
        if (!this.eventBus || this.destroyed) {
            return
        }

        try {
            this.eventBus.emit(eventName, payload)
        } catch (error) {
            console.error(
                `[PerformanceSystem] Ошибка события ${eventName}:`,
                error
            )
        }
    }

    // --------------------------------------------------------
    // DESTROY
    // --------------------------------------------------------

    destroy() {
        if (this.destroyed) {
            return
        }

        this.emit('performance:destroyed')

        this.enabled = false
        this.paused = true

        this.frameTimes = []
        this.frameHistory = []

        this.systemMetrics.clear()

        this.lastMemory = null
        this.lastWarning = null

        this.eventBus = null

        this.initialized = false
        this.destroyed = true
    }
}

// ============================================================
// SAYLO GAME ENGINE
// SYSTEM: RENDERING SYSTEM
// Управление отрисовкой игрового мира
// PixiJS 8.x
// Версия архитектуры: 2.0
// ============================================================

export class RenderingSystem {

    constructor({
        app,
        world,
        camera,
        eventBus = null,
        config = {}
    } = {}) {

        if (!app || !app.renderer || !app.stage) {
            throw new TypeError(
                '[RenderingSystem] Требуется инициализированный Pixi Application.'
            )
        }

        if (!world || !world.container) {
            throw new TypeError(
                '[RenderingSystem] Требуется корректный World.'
            )
        }

        if (
            !camera ||
            typeof camera.resize !== 'function'
        ) {
            throw new TypeError(
                '[RenderingSystem] Требуется корректная Camera.'
            )
        }

        if (
            eventBus !== null &&
            typeof eventBus.emit !== 'function'
        ) {
            throw new TypeError(
                '[RenderingSystem] Некорректный EventBus.'
            )
        }

        this.app = app
        this.renderer = app.renderer
        this.stage = app.stage

        this.world = world
        this.camera = camera
        this.eventBus = eventBus

        this.config = {
            enabled: true,
            autoRender: true,
            targetFPS: 60,
            maxDeltaTime: 0.05,
            pauseWhenHidden: true,
            clearBeforeRender: true,
            ...config
        }

        this.initialized = false
        this.destroyed = false
        this.paused = false

        this.frameCount = 0
        this.elapsedTime = 0
        this.frameAccumulator = 0
        this.lastRenderTime = 0

        this.frameInterval = this.config.targetFPS > 0
            ? 1 / this.config.targetFPS
            : 0

        this.viewport = {
            width: 0,
            height: 0
        }

        this.stats = {
            frames: 0,
            totalRenderTime: 0,
            averageRenderTime: 0,
            lastRenderTime: 0,
            skippedFrames: 0
        }

        this.resizeObserver = null
        this.handleResize = this.resize.bind(this)
        this.handleVisibilityChange =
            this.onVisibilityChange.bind(this)

        this.initialize()
    }

    // --------------------------------------------------------
    // INITIALIZATION
    // --------------------------------------------------------

    initialize() {

        this.ensureActive()

        if (this.initialized) {
            return true
        }

        const screen = this.renderer.screen

        this.viewport.width = screen.width
        this.viewport.height = screen.height

        this.camera.resize(
            screen.width,
            screen.height
        )

        window.addEventListener(
            'resize',
            this.handleResize
        )

        // SAYLO RESIZE SYNC V1
        if (typeof ResizeObserver !== 'undefined') {
            const resizeTarget = this.app.canvas?.parentElement

            if (resizeTarget) {
                this.resizeObserver = new ResizeObserver(() => {
                    if (this.destroyed) return
                    this.resize()
                })

                this.resizeObserver.observe(resizeTarget)
            }
        }

        if (this.config.pauseWhenHidden) {
            document.addEventListener(
                'visibilitychange',
                this.handleVisibilityChange
            )
        }

        this.initialized = true

        this.emit('rendering:initialized', {
            width: screen.width,
            height: screen.height
        })

        return true
    }

    // --------------------------------------------------------
    // MAIN UPDATE
    // Вызывается игровым циклом.
    // deltaTime передаётся в секундах.
    // --------------------------------------------------------

    update(deltaTime = 0) {

        this.ensureActive()

        if (
            !this.initialized ||
            !this.config.enabled ||
            this.paused ||
            !this.config.autoRender
        ) {
            return false
        }

        if (
            !Number.isFinite(deltaTime) ||
            deltaTime < 0
        ) {
            throw new TypeError(
                '[RenderingSystem] Некорректный deltaTime.'
            )
        }

        const safeDelta = Math.min(
            deltaTime,
            this.config.maxDeltaTime
        )

        this.frameAccumulator += safeDelta

        if (
            this.frameInterval > 0 &&
            this.frameAccumulator < this.frameInterval
        ) {
            this.stats.skippedFrames++
            return false
        }

        this.frameAccumulator = this.frameInterval > 0
            ? this.frameAccumulator % this.frameInterval
            : 0

        return this.render()
    }

    // --------------------------------------------------------
    // RENDER
    // --------------------------------------------------------

    render() {

        this.ensureActive()

        if (
            !this.initialized ||
            !this.config.enabled ||
            this.paused
        ) {
            return false
        }

        const startTime = performance.now()

        try {

            this.renderer.render({
                container: this.stage,
                clear: this.config.clearBeforeRender
            })

            const renderTime =
                performance.now() - startTime

            this.frameCount++
            this.elapsedTime += renderTime / 1000

            this.lastRenderTime = renderTime

            this.stats.frames = this.frameCount
            this.stats.lastRenderTime = renderTime

            this.stats.totalRenderTime += renderTime

            this.stats.averageRenderTime =
                this.stats.totalRenderTime /
                this.frameCount

            this.emit('rendering:frame', {
                frame: this.frameCount,
                renderTime,
                averageRenderTime:
                    this.stats.averageRenderTime
            })

            return true

        } catch (error) {

            this.emit('rendering:error', {
                error,
                frame: this.frameCount
            })

            console.error(
                '[RenderingSystem] Ошибка отрисовки:',
                error
            )

            return false
        }
    }

    // --------------------------------------------------------
    // RESIZE
    // --------------------------------------------------------

    resize() {

        this.ensureActive()

        if (!this.app || !this.renderer) {
            return false
        }

        // Берём фактический размер игрового контейнера.
        // Не полагаемся на устаревший размер PixiJS.

        const resizeTarget = this.app.canvas?.parentElement
        const bounds = resizeTarget?.getBoundingClientRect()

        const width = bounds?.width > 0
            ? bounds.width
            : this.renderer.screen.width

        const height = bounds?.height > 0
            ? bounds.height
            : this.renderer.screen.height

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width <= 0 ||
            height <= 0
        ) {
            return false
        }

        // Сначала обновляем renderer.
        const currentScreen = this.renderer.screen

        if (
            Math.abs(currentScreen.width - width) > 0.5 ||
            Math.abs(currentScreen.height - height) > 0.5
        ) {
            this.renderer.resize(width, height)
        }

        // Затем синхронизируем камеру с реальным экраном PixiJS.
        const screen = this.renderer.screen

        if (
            Math.abs(this.camera.viewportWidth - screen.width) <= 0.5 &&
            Math.abs(this.camera.viewportHeight - screen.height) <= 0.5
        ) {
            return true
        }

        this.viewport.width = screen.width
        this.viewport.height = screen.height

        this.camera.resize(
            screen.width,
            screen.height
        )

        this.emit('rendering:resize', {
            width: screen.width,
            height: screen.height
        })

        if (this.config.autoRender) {
            this.render()
        }

        return true
    }

    // --------------------------------------------------------
    // VISIBILITY
    // --------------------------------------------------------

    onVisibilityChange() {

        if (!this.config.pauseWhenHidden) {
            return
        }

        if (document.hidden) {
            this.pause()
        } else {
            this.resume()
            this.render()
        }
    }

    // --------------------------------------------------------
    // CONTROL
    // --------------------------------------------------------

    pause() {

        this.ensureActive()

        if (this.paused) {
            return false
        }

        this.paused = true
        this.frameAccumulator = 0

        this.emit('rendering:paused')

        return true
    }

    resume() {

        this.ensureActive()

        if (!this.paused) {
            return false
        }

        this.paused = false
        this.frameAccumulator = 0

        this.emit('rendering:resumed')

        return true
    }

    setEnabled(enabled) {

        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError(
                '[RenderingSystem] enabled должен быть boolean.'
            )
        }

        this.config.enabled = enabled

        if (!enabled) {
            this.frameAccumulator = 0
        }

        this.emit('rendering:enabled-changed', {
            enabled
        })

        return true
    }

    setAutoRender(enabled) {

        this.ensureActive()

        if (typeof enabled !== 'boolean') {
            throw new TypeError(
                '[RenderingSystem] autoRender должен быть boolean.'
            )
        }

        this.config.autoRender = enabled
        this.frameAccumulator = 0

        return true
    }

    // --------------------------------------------------------
    // FRAME RATE
    // --------------------------------------------------------

    setTargetFPS(fps) {

        this.ensureActive()

        if (
            !Number.isFinite(fps) ||
            fps < 0 ||
            fps > 240
        ) {
            throw new RangeError(
                '[RenderingSystem] FPS должен быть от 0 до 240.'
            )
        }

        this.config.targetFPS = fps

        this.frameInterval = fps > 0
            ? 1 / fps
            : 0

        this.frameAccumulator = 0

        this.emit('rendering:fps-changed', {
            targetFPS: fps
        })

        return true
    }

    // --------------------------------------------------------
    // STATISTICS
    // --------------------------------------------------------

    getStats() {

        return Object.freeze({
            ...this.stats,
            frameCount: this.frameCount,
            elapsedTime: this.elapsedTime,
            targetFPS: this.config.targetFPS,
            enabled: this.config.enabled,
            paused: this.paused,
            viewport: Object.freeze({
                ...this.viewport
            })
        })
    }

    resetStats() {

        this.ensureActive()

        this.frameCount = 0
        this.elapsedTime = 0
        this.lastRenderTime = 0

        this.stats = {
            frames: 0,
            totalRenderTime: 0,
            averageRenderTime: 0,
            lastRenderTime: 0,
            skippedFrames: 0
        }

        this.emit('rendering:stats-reset')

        return true
    }

    // --------------------------------------------------------
    // EVENTS
    // --------------------------------------------------------

    emit(eventName, payload = {}) {

        if (!this.eventBus) {
            return false
        }

        return this.eventBus.emit(
            eventName,
            payload
        )
    }

    // --------------------------------------------------------
    // VALIDATION
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[RenderingSystem] Система уничтожена.'
            )
        }
    }

    // --------------------------------------------------------
    // DESTROY
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) {
            return
        }

        if (this.resizeObserver) {
            this.resizeObserver.disconnect()
            this.resizeObserver = null
        }

        if (this.initialized) {

            window.removeEventListener(
                'resize',
                this.handleResize
            )

            document.removeEventListener(
                'visibilitychange',
                this.handleVisibilityChange
            )
        }

        this.emit('rendering:destroyed')

        this.initialized = false
        this.paused = true

        this.app = null
        this.renderer = null
        this.stage = null

        this.world = null
        this.camera = null
        this.eventBus = null

        this.handleResize = null
        this.handleVisibilityChange = null

        this.destroyed = true
    }
}

// SAYLO GAME ENGINE

// Версия архитектуры: 1.0
// ============================================================

export class Camera {

    constructor({
        world,
        viewportWidth = 1920,
        viewportHeight = 1080,
        scaleMode = 'contain',
        minZoom = 0.5,
        maxZoom = 2
    } = {}) {

        if (!world || !world.container) {
            throw new TypeError(
                '[Saylo Camera] Необходимо передать игровой мир.'
            )
        }

        this.validateViewport(viewportWidth, viewportHeight)

        if (!['contain', 'cover'].includes(scaleMode)) {
            throw new Error(
                '[Saylo Camera] Неизвестный режим масштабирования.'
            )
        }

        if (
            !Number.isFinite(minZoom) ||
            !Number.isFinite(maxZoom) ||
            minZoom <= 0 ||
            maxZoom < minZoom
        ) {
            throw new RangeError(
                '[Saylo Camera] Некорректные ограничения масштаба.'
            )
        }

        this.world = world

        this.viewportWidth = viewportWidth
        this.viewportHeight = viewportHeight

        this.scaleMode = scaleMode

        this.minZoom = minZoom
        this.maxZoom = maxZoom

        this.zoom = 1
        this.baseScale = 1
        this.scale = 1

        // Камера хранит координаты центра видимой области мира.

        this.x = world.width / 2
        this.y = world.height / 2

        this.destroyed = false

        this.calculateBaseScale()
        this.applyTransform()

        console.log('[Saylo Camera] Камера создана.')
    }

    // --------------------------------------------------------
    // РАСЧЁТ БАЗОВОГО МАСШТАБА
    // --------------------------------------------------------

    calculateBaseScale() {

        const scaleX = this.viewportWidth / this.world.width
        const scaleY = this.viewportHeight / this.world.height

        this.baseScale = this.scaleMode === 'contain'
            ? Math.min(scaleX, scaleY)
            : Math.max(scaleX, scaleY)

        this.scale = this.baseScale * this.zoom
    }

    // --------------------------------------------------------
    // ПРИМЕНЕНИЕ ТРАНСФОРМАЦИИ К МИРУ
    // --------------------------------------------------------

    applyTransform() {

        this.ensureActive()

        this.scale = this.baseScale * this.zoom

        this.clampPosition()

        this.world.container.scale.set(this.scale)

        this.world.container.position.set(
            this.viewportWidth / 2 - this.x * this.scale,
            this.viewportHeight / 2 - this.y * this.scale
        )
    }

    // --------------------------------------------------------
    // ПЕРЕМЕЩЕНИЕ КАМЕРЫ
    // --------------------------------------------------------

    setPosition(x, y) {

        this.ensureActive()

        if (
            !Number.isFinite(x) ||
            !Number.isFinite(y)
        ) {
            throw new TypeError(
                '[Saylo Camera] Координаты должны быть числами.'
            )
        }

        this.x = x
        this.y = y

        this.applyTransform()
    }

    // --------------------------------------------------------
    // СМЕЩЕНИЕ КАМЕРЫ
    // --------------------------------------------------------

    moveBy(deltaX, deltaY) {

        this.ensureActive()

        if (
            !Number.isFinite(deltaX) ||
            !Number.isFinite(deltaY)
        ) {
            throw new TypeError(
                '[Saylo Camera] Смещение должно быть числовым.'
            )
        }

        this.setPosition(
            this.x + deltaX,
            this.y + deltaY
        )
    }

    // --------------------------------------------------------
    // ЦЕНТРИРОВАНИЕ НА ТОЧКЕ
    // --------------------------------------------------------

    lookAt(x, y) {

        this.setPosition(x, y)
    }

    // --------------------------------------------------------
    // ИЗМЕНЕНИЕ МАСШТАБА
    // --------------------------------------------------------

    setZoom(value) {

        this.ensureActive()

        if (!Number.isFinite(value) || value <= 0) {
            throw new RangeError(
                '[Saylo Camera] Масштаб должен быть положительным числом.'
            )
        }

        this.zoom = Math.min(
            this.maxZoom,
            Math.max(this.minZoom, value)
        )

        this.applyTransform()
    }

    // --------------------------------------------------------
    // ИЗМЕНЕНИЕ РАЗМЕРА ЭКРАННОЙ ОБЛАСТИ
    // --------------------------------------------------------

    resize(width, height) {

        this.ensureActive()

        this.validateViewport(width, height)

        this.viewportWidth = width
        this.viewportHeight = height

        this.calculateBaseScale()
        this.applyTransform()
    }

    // --------------------------------------------------------
    // ОГРАНИЧЕНИЕ КАМЕРЫ ГРАНИЦАМИ МИРА
    // --------------------------------------------------------

    clampPosition() {

        const visibleWidth = this.viewportWidth / this.scale
        const visibleHeight = this.viewportHeight / this.scale

        const halfWidth = visibleWidth / 2
        const halfHeight = visibleHeight / 2

        if (visibleWidth >= this.world.width) {
            this.x = this.world.width / 2
        } else {
            this.x = Math.max(
                halfWidth,
                Math.min(this.world.width - halfWidth, this.x)
            )
        }

        if (visibleHeight >= this.world.height) {
            this.y = this.world.height / 2
        } else {
            this.y = Math.max(
                halfHeight,
                Math.min(this.world.height - halfHeight, this.y)
            )
        }
    }

    // --------------------------------------------------------
    // ПРЕОБРАЗОВАНИЕ КООРДИНАТ МИРА В ЭКРАННЫЕ
    // --------------------------------------------------------

    worldToScreen(x, y) {

        this.ensureActive()

        return {
            x: this.viewportWidth / 2 + (x - this.x) * this.scale,
            y: this.viewportHeight / 2 + (y - this.y) * this.scale
        }
    }

    // --------------------------------------------------------
    // ПРЕОБРАЗОВАНИЕ ЭКРАННЫХ КООРДИНАТ В КООРДИНАТЫ МИРА
    // --------------------------------------------------------

    screenToWorld(x, y) {

        this.ensureActive()

        return {
            x: this.x + (x - this.viewportWidth / 2) / this.scale,
            y: this.y + (y - this.viewportHeight / 2) / this.scale
        }
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ ПАРАМЕТРОВ КАМЕРЫ
    // --------------------------------------------------------

    getState() {

        this.ensureActive()

        return Object.freeze({

            x: this.x,
            y: this.y,

            zoom: this.zoom,
            scale: this.scale,

            viewportWidth: this.viewportWidth,
            viewportHeight: this.viewportHeight
        })
    }

    // --------------------------------------------------------
    // ПРОВЕРКА РАЗМЕРОВ ЭКРАННОЙ ОБЛАСТИ
    // --------------------------------------------------------

    validateViewport(width, height) {

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width <= 0 ||
            height <= 0
        ) {
            throw new RangeError(
                '[Saylo Camera] Размеры области должны быть положительными.'
            )
        }
    }

    // --------------------------------------------------------
    // ПРОВЕРКА ДОСТУПНОСТИ
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[Saylo Camera] Камера уничтожена.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ КАМЕРЫ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.world = null

        this.viewportWidth = 0
        this.viewportHeight = 0

        this.x = 0
        this.y = 0

        this.scale = 1
        this.baseScale = 1
        this.zoom = 1

        this.destroyed = true

        console.log('[Saylo Camera] Камера уничтожена.')
    }
}

export class CoordinateSystem {

    constructor(camera = null) {

        this.camera = camera
        this.destroyed = false
    }

    // --------------------------------------------------------
    // КООРДИНАТЫ МИРА -> ЭКРАН
    // --------------------------------------------------------

    worldToScreen(x, y) {

        this.ensureActive()
        this.validatePoint(x, y)
        this.ensureCamera()

        return this.camera.worldToScreen(x, y)
    }

    // --------------------------------------------------------
    // ЭКРАН -> КООРДИНАТЫ МИРА
    // --------------------------------------------------------

    screenToWorld(x, y) {

        this.ensureActive()
        this.validatePoint(x, y)
        this.ensureCamera()

        return this.camera.screenToWorld(x, y)
    }

    // --------------------------------------------------------
    // РАССТОЯНИЕ МЕЖДУ ДВУМЯ ТОЧКАМИ
    // --------------------------------------------------------

    distance(x1, y1, x2, y2) {

        this.ensureActive()

        this.validatePoint(x1, y1)
        this.validatePoint(x2, y2)

        return Math.hypot(
            x2 - x1,
            y2 - y1
        )
    }

    // --------------------------------------------------------
    // ЦЕНТР ПРЯМОУГОЛЬНОЙ ОБЛАСТИ
    // --------------------------------------------------------

    getCenter(x, y, width, height) {

        this.ensureActive()
        this.validateRectangle(x, y, width, height)

        return {
            x: x + width / 2,
            y: y + height / 2
        }
    }

    // --------------------------------------------------------
    // ГРАНИЦЫ ПРЯМОУГОЛЬНОЙ ОБЛАСТИ
    // --------------------------------------------------------

    getBounds(x, y, width, height) {

        this.ensureActive()
        this.validateRectangle(x, y, width, height)

        return {
            left: x,
            top: y,
            right: x + width,
            bottom: y + height,
            width,
            height
        }
    }

    // --------------------------------------------------------
    // ПРОВЕРКА НАХОЖДЕНИЯ ТОЧКИ В ПРЯМОУГОЛЬНИКЕ
    // --------------------------------------------------------

    containsPoint(pointX, pointY, x, y, width, height) {

        this.ensureActive()

        this.validatePoint(pointX, pointY)
        this.validateRectangle(x, y, width, height)

        return (
            pointX >= x &&
            pointX <= x + width &&
            pointY >= y &&
            pointY <= y + height
        )
    }

    // --------------------------------------------------------
    // ПРОВЕРКА КООРДИНАТ
    // --------------------------------------------------------

    validatePoint(x, y) {

        if (
            !Number.isFinite(x) ||
            !Number.isFinite(y)
        ) {
            throw new TypeError(
                '[CoordinateSystem] Координаты должны быть конечными числами.'
            )
        }
    }

    validateRectangle(x, y, width, height) {

        this.validatePoint(x, y)

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width < 0 ||
            height < 0
        ) {
            throw new TypeError(
                '[CoordinateSystem] Некорректные размеры области.'
            )
        }
    }

    // --------------------------------------------------------
    // ПРОВЕРКА КАМЕРЫ
    // --------------------------------------------------------

    ensureCamera() {

        if (
            !this.camera ||
            typeof this.camera.worldToScreen !== 'function' ||
            typeof this.camera.screenToWorld !== 'function'
        ) {
            throw new Error(
                '[CoordinateSystem] Камера не подключена или не поддерживает преобразование координат.'
            )
        }
    }

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[CoordinateSystem] Модуль уничтожен.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.camera = null
        this.destroyed = true
    }
}

// SAYLO GAME ENGINE

import { Container } from 'pixi.js'

export class World {

    constructor({
        width = 1920,
        height = 1080
    } = {}) {

        this.validateSize(width, height)

        this.width = width
        this.height = height

        this.destroyed = false
        this.updatables = new Set()

        // ----------------------------------------------------
        // КОРНЕВОЙ КОНТЕЙНЕР МИРА
        // ----------------------------------------------------

        this.container = new Container()
        this.container.label = 'SayloWorld'

        // ----------------------------------------------------
        // СЛОИ ИГРОВОГО МИРА
        // Порядок добавления определяет порядок отрисовки.
        // ----------------------------------------------------

        this.layers = {

            background: this.createLayer('Background'),

            environment: this.createLayer('Environment'),

            objects: this.createLayer('Objects'),

            characters: this.createLayer('Characters'),

            effects: this.createLayer('Effects'),

            foreground: this.createLayer('Foreground')
        }

        for (const layer of Object.values(this.layers)) {
            this.container.addChild(layer)
        }

        console.log('[Saylo World] Игровой мир создан.')
    }

    // --------------------------------------------------------
    // СОЗДАНИЕ СЛОЯ
    // --------------------------------------------------------

    createLayer(name) {

        const layer = new Container()

        layer.label = `World_${name}`

        return layer
    }

    // --------------------------------------------------------
    // ПОДКЛЮЧЕНИЕ МИРА К PIXI STAGE
    // --------------------------------------------------------

    mount(stage) {

        this.ensureActive()

        if (!stage || typeof stage.addChild !== 'function') {
            throw new TypeError(
                '[Saylo World] Передан некорректный PixiJS stage.'
            )
        }

        if (this.container.parent === stage) {
            return false
        }

        if (this.container.parent) {
            throw new Error(
                '[Saylo World] Мир уже подключён к другому контейнеру.'
            )
        }

        stage.addChild(this.container)

        return true
    }

    // --------------------------------------------------------
    // ОТКЛЮЧЕНИЕ МИРА ОТ PIXI STAGE
    // --------------------------------------------------------

    unmount() {

        this.ensureActive()

        if (!this.container.parent) {
            return false
        }

        this.container.parent.removeChild(this.container)

        return true
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ СЛОЯ
    // --------------------------------------------------------

    getLayer(name) {

        this.ensureActive()

        const layer = this.layers[name]

        if (!layer) {
            throw new Error(
                `[Saylo World] Неизвестный слой: ${name}`
            )
        }

        return layer
    }

    // --------------------------------------------------------
    // ДОБАВЛЕНИЕ ОБЪЕКТА В МИР
    // --------------------------------------------------------

    addToLayer(displayObject, layerName = 'objects') {

        this.ensureActive()

        if (
            !displayObject ||
            typeof displayObject !== 'object' ||
            typeof displayObject.destroy !== 'function'
        ) {
            throw new TypeError(
                '[Saylo World] Ожидается объект отображения PixiJS.'
            )
        }

        const layer = this.getLayer(layerName)

        if (displayObject.parent === layer) {
            return false
        }

        if (displayObject.parent) {
            throw new Error(
                '[Saylo World] Объект уже принадлежит другому контейнеру.'
            )
        }

        layer.addChild(displayObject)

        return true
    }

    // --------------------------------------------------------
    // УДАЛЕНИЕ ОБЪЕКТА ИЗ МИРА
    // --------------------------------------------------------

    removeFromLayer(displayObject) {

        this.ensureActive()

        if (!displayObject || !displayObject.parent) {
            return false
        }

        const belongsToWorld = Object.values(this.layers).some(
            layer => displayObject.parent === layer
        )

        if (!belongsToWorld) {
            return false
        }

        displayObject.parent.removeChild(displayObject)

        return true
    }

    // --------------------------------------------------------
    // РЕГИСТРАЦИЯ ОБНОВЛЯЕМЫХ ОБЪЕКТОВ
    // --------------------------------------------------------

    addUpdatable(object) {

        this.ensureActive()

        if (!object || typeof object.update !== 'function') {
            throw new TypeError(
                '[Saylo World] Объект должен иметь метод update().'
            )
        }

        this.updatables.add(object)

        return true
    }

    // --------------------------------------------------------
    // УДАЛЕНИЕ ОБНОВЛЯЕМОГО ОБЪЕКТА
    // --------------------------------------------------------

    removeUpdatable(object) {

        this.ensureActive()

        return this.updatables.delete(object)
    }

    // --------------------------------------------------------
    // ОБНОВЛЕНИЕ ИГРОВОГО МИРА
    // --------------------------------------------------------

    update(deltaTime, elapsedTime = 0) {

        this.ensureActive()

        if (
            !Number.isFinite(deltaTime) ||
            deltaTime < 0
        ) {
            throw new RangeError(
                '[Saylo World] Некорректное deltaTime.'
            )
        }

        for (const object of [...this.updatables]) {

            if (this.updatables.has(object)) {
                object.update(deltaTime, elapsedTime)
            }
        }
    }

    // --------------------------------------------------------
    // ИЗМЕНЕНИЕ РАЗМЕРОВ МИРА
    // --------------------------------------------------------

    setSize(width, height) {

        this.ensureActive()

        this.validateSize(width, height)

        this.width = width
        this.height = height
    }

    // --------------------------------------------------------
    // ПОЛУЧЕНИЕ ГРАНИЦ МИРА
    // --------------------------------------------------------

    getBounds() {

        this.ensureActive()

        return Object.freeze({

            x: 0,
            y: 0,

            width: this.width,
            height: this.height
        })
    }

    // --------------------------------------------------------
    // ПРОВЕРКА РАЗМЕРОВ
    // --------------------------------------------------------

    validateSize(width, height) {

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(height) ||
            width <= 0 ||
            height <= 0
        ) {
            throw new RangeError(
                '[Saylo World] Ширина и высота должны быть положительными числами.'
            )
        }
    }

    // --------------------------------------------------------
    // ПРОВЕРКА ДОСТУПНОСТИ
    // --------------------------------------------------------

    ensureActive() {

        if (this.destroyed) {
            throw new Error(
                '[Saylo World] Игровой мир уничтожен.'
            )
        }
    }

    // --------------------------------------------------------
    // УНИЧТОЖЕНИЕ МИРА
    // --------------------------------------------------------

    destroy() {

        if (this.destroyed) return

        this.updatables.clear()

        if (this.container.parent) {
            this.container.parent.removeChild(this.container)
        }

        this.container.destroy({
            children: true
        })

        this.layers = null
        this.container = null

        this.width = 0
        this.height = 0

        this.destroyed = true

        console.log('[Saylo World] Игровой мир уничтожен.')
    }
}
